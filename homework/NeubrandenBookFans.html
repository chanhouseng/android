var startX = 0f
var startY = 0f
findViewById<FrameLayout>(R.id.frameLayout).setOnTouchListener { v, event ->
    when(event.action){
        MotionEvent.ACTION_DOWN -> {
            startY = event.y
            startX = event.x
        }

        MotionEvent.ACTION_UP -> {
            val moveX = event.x
            val moveY = event.y
            val deltaX = moveX - startX
            val deltaY = moveY - startY
            if(Math.abs(deltaX) > Math.abs(deltaY)){
                if(deltaX > 0){
                    countDownTimer.cancel()
                    if(now == 0){
                        now = Data.jsonArray.books.size -1
                    }
                    else{
                        now --
                    }
                    countDownTimer.start()      //因為用一個countDownTimer，就可以將佢地啲野寫一次，之後調用countDownTimer即可。同埋佢一開始就會跳一下，所以就可以用到
                }
                else{
                    countDownTimer.cancel()
                    if(now == Data.jsonArray.books.size-1){
                        now = 0
                    }
                    else{
                        now ++
                    }
                    countDownTimer.start()
                }
            }
        }
    }
    true
}

countDownTimer = object : CountDownTimer(5000, 5000){
    override fun onFinish() {
        if(now != Data.jsonArray.books.size -1){     //不同唔使咁樣，其實用%都可以。就now = (now + 1) % Data.jsonArray.books.size
            now++
        }
        else{
            now = 0
        }
        start()
    }

    override fun onTick(millisUntilFinished: Long) {
        val book = Data.jsonArray.books[now]
        imageView3.setImageResource(resources.getIdentifier(book.id.replace("-", "_").replace("ä", "_").replace("ú", "_"), "drawable", packageName))
        textView3.text = book.title
        textView4.text = book.author
        textView5.text = book.description
        textView9.text = "(${book.reviews} reviews)"
        textView6.text = book.rating.toString()
        button.setOnClickListener {
            val intent = Intent(this@MainActivity, MainActivity2::class.java)
            intent.putExtra("id", book.id)
            startActivity(intent)
        }
    }
}

fun createAnim(scrollView: HorizontalScrollView, number: Int){          number嘅作用係分清佢係向左定向右，咁就唔使寫兩個
    val valeuAnimator = ValueAnimator.ofInt(scrollView.scrollX, scrollView.scrollX + number)
    valeuAnimator.duration = 400
    valeuAnimator.interpolator = LinearInterpolator()
    valeuAnimator.addUpdateListener {
        scrollView.scrollX = it.animatedValue as Int
    }
    valeuAnimator.start()
}

fun loadCustom(custom: View){
    custom.findViewById<TextView>(R.id.textView34).text = "Preview - Page ${reviewNumber+1} of ${book.totalPages}"
    if(book.previewPages != null && reviewNumber < book.previewPages!!.size){    //可能唔俾previews或者已經超過previews的上限
        custom.findViewById<TextView>(R.id.textView35).text = book.previewPages?.get(reviewNumber)
    }
    custom.findViewById<LinearLayout>(R.id.pageLayout).removeAllViews()
    var i = 0
    while(i <= book.totalPages-1){
        val layout = layoutInflater.inflate(R.layout.page_pro, null)
        if(book.previewPages != null){
            if(i < book.previewPages!!.size){     //可以previews的範圍
                layout.findViewById<LinearLayout>(R.id.canView).visibility = View.VISIBLE
                layout.findViewById<LinearLayout>(R.id.canotView).visibility = View.GONE
                layout.findViewById<TextView>(R.id.textView37).text = (i+1).toString()
                if (i == reviewNumber){
                    layout.findViewById<LinearLayout>(R.id.canView).setBackgroundResource(R.drawable.cliuck_page_back)
                    custom.findViewById<CardView>(R.id.alertLayout).visibility = View.GONE
                    custom.findViewById<TextView>(R.id.textView36).setTextColor(Color.parseColor("#248155"))
                    custom.findViewById<TextView>(R.id.textView36).text = "Preview Available"
                }
            }
            else{       //不可以previews的範圍
                layout.findViewById<LinearLayout>(R.id.canotView).visibility = View.VISIBLE
                layout.findViewById<LinearLayout>(R.id.canView).visibility = View.GONE
                if (i == reviewNumber){
                    layout.findViewById<LinearLayout>(R.id.canotView).setBackgroundResource(R.drawable.click_no_back)
                    custom.findViewById<CardView>(R.id.alertLayout).visibility = View.VISIBLE
                    custom.findViewById<TextView>(R.id.textView36).setTextColor(Color.parseColor("#b8742a"))
                    custom.findViewById<TextView>(R.id.textView36).text = "Purchase Required"
                }
            }
        }
        else{          //沒有preview pages
            layout.findViewById<LinearLayout>(R.id.canotView).visibility = View.VISIBLE
            layout.findViewById<LinearLayout>(R.id.canView).visibility = View.GONE
            if (i == reviewNumber){
                layout.findViewById<LinearLayout>(R.id.canotView).setBackgroundResource(R.drawable.click_no_back)
                custom.findViewById<CardView>(R.id.alertLayout).visibility = View.VISIBLE
                custom.findViewById<TextView>(R.id.textView36).setTextColor(Color.parseColor("#b8742a"))
                custom.findViewById<TextView>(R.id.textView36).text = "Purchase Required"
            }
        }
        var kI = i
        layout.setOnClickListener {
            reviewNumber = kI      //雖然會令APP變慢，但係對於比賽係個好方法，唔使改好多野，喺前面寫曬就得，就能夠實現即時更新
            loadCustom(custom)
        }
        custom.findViewById<LinearLayout>(R.id.pageLayout).addView(layout)
        i++
    }
}
