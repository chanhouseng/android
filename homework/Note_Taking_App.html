        val newChild = children?.map { it.deepCopy() }?.toMutableList() as? ArrayList<ContentClass> ?: arrayListOf()
        
        return ContentClass(
            type,imageId, text, newChild, isChecked, id
        )

        //代入一層可理解，先將佢啲children 所有項都deepCopy，即係不斷地令佢(ContentClass)的children都做同一樣野
        //做啲咩?因為佢係.map，本身就係指緊children呢個項，就要將choldren複製arrayList<ContentClass>，不成功就變arrayList空數組
        //成功複製之後就回傳返成個ContentClass，其他值不變，只變chilren
    }
}

fun createImageUri(): Uri {    //在手機的「Pictures」資料夾中建立一個新的 JPEG 圖片檔案，並取得它的 Uri 路徑
    val contentValues = ContentValues().apply {   //其實同儲存圖片差唔多，可以理解係我要set好啲相片的資料，之後再返回一個uri以表示將佢嘅資料
        put(MediaStore.Images.Media.DISPLAY_NAME, "photo_${System.currentTimeMillis()}.jpg")
        put(MediaStore.Images.Media.MIME_TYPE, "image/jpeg")
        put(MediaStore.Images.Media.RELATIVE_PATH, Environment.DIRECTORY_PICTURES)
    }
    return contentResolver.insert(MediaStore.Images.Media.EXTERNAL_CONTENT_URI, contentValues)!!
}

fun load(){
    var i = 0
    linearLayout.removeAllViews()
    for(result in note.content){
        result.id = "${i}"    //第一層ID
        if(result.type == "image"){     //妢係唔同類型有唔同操作
            val imageView = ImageView(this)
            if(result.imageId?.contains("content://") == true){     //分係uri定預設檔名，不過可以試下用try
                imageView.setImageURI(result.imageId?.toUri())
            }
            else if(result.imageId?.any{it.isDigit()} ?: false){
                imageView.setBackgroundResource(R.drawable.cupcake)
            }
            imageView.layoutParams = ViewGroup.LayoutParams(WindowManager.LayoutParams.MATCH_PARENT,    
                1000)
            imageView.scaleType = ImageView.ScaleType.CENTER_CROP
            linearLayout.addView(imageView)
        }
        else if(result.type == "text"){
            val editText = EditText(this)
            editText.setText(result.text)
            editText.setOnFocusChangeListener{_, hasFocus ->
                if(hasFocus){
                    nowId = result.id
                    println(nowId)
                }
            }
            editText.doOnTextChanged { text, start, before, count ->
                if(isRestoring){
                    return@doOnTextChanged
                }
                result.text = text.toString()
                saveHost()
            }
            linearLayout.addView(editText)
        }
        else if(result.type == "listItem"){
            val layout = layoutInflater.inflate(R.layout.list_item_pro, null)
            layout.findViewById<EditText>(R.id.editTextText3).setText(result.text)
            if(!result.children.isNullOrEmpty()){     //如果佢仲有子元素，就要繼續load落去，進入遞歸函數
                loadListItem(result.id,result.children!!, layout)
            }
            layout.findViewById<EditText>(R.id.editTextText3).setOnFocusChangeListener{_, hasFocus ->
                if(hasFocus){
                    nowId = result.id
                }
            }
            layout.findViewById<EditText>(R.id.editTextText3).doOnTextChanged { text, start, before, count ->
                if(isRestoring){
                    return@doOnTextChanged    //都係setText會觸發listener，以防止此事所以有限制
                }
                result.text = text.toString()
                saveHost()
            }
            linearLayout.addView(layout)
        }
        else if(result.type == "checkboxListItem"){
            val layout = layoutInflater.inflate(R.layout.check_box_pro, null)
            layout.findViewById<EditText>(R.id.editTextText3).setText(result.text)
            if(result.isChecked == true){
                layout.findViewById<CheckBox>(R.id.checkBox).isChecked = true
            }
            if(!result.children.isNullOrEmpty()){
                loadListItem(result.id, result.children!!, layout)
            }
            layout.findViewById<EditText>(R.id.editTextText3).setOnFocusChangeListener{_, hasFocus ->
                if(hasFocus){
                    nowId = result.id
                }
            }
            layout.findViewById<EditText>(R.id.editTextText3).doOnTextChanged { text, start, before, count ->
                if(isRestoring){
                    return@doOnTextChanged
                }
                result.text = text.toString()
                saveHost()
            }
            linearLayout.addView(layout)
        }
        i++
    }
}

fun loadListItem(resultId: String?, children: ArrayList<ContentClass>, layout: View){    
    var k = 0
    for(children1 in children){
        children1.id = resultId    //佢要繼承返父的ID
        children1.id += "-${k}"    //都要自己加返啲ID
        if(children1.type == "image"){
            val imageView = ImageView(this)
            if(children1.imageId?.contains("content://") == true){
                imageView.setImageURI(children1.imageId?.toUri())
            }
            else if(children1.imageId?.any{it.isDigit()} ?: false){
                imageView.setBackgroundResource(R.drawable.cupcake)
            }
            imageView.layoutParams = ViewGroup.LayoutParams(WindowManager.LayoutParams.MATCH_PARENT,
                1000)
            imageView.scaleType = ImageView.ScaleType.CENTER_CROP
            layout.findViewById<LinearLayout>(R.id.linearLayout).addView(imageView)    //要喺父加入view，如果唔係就會斷左
        }
        else if(children1.type == "text"){
            val editText = EditText(this)
            editText.setText(children1.text)
            editText.setOnFocusChangeListener{_, hasFocus ->
                if(hasFocus){
                    nowId = children1.id
                }
            }
            editText.doOnTextChanged { text, start, before, count ->
                if(isRestoring) return@doOnTextChanged
                children1.text = text.toString()
                saveHost()
            }
            layout.findViewById<LinearLayout>(R.id.linearLayout).addView(editText)
        }
        else if (children1.type == "listItem"){

            if(!children1.children.isNullOrEmpty()){
                loadListItem(children1.id, children1.children!!, layout2)    //因為宜家layout2變左父，佢啲子代要加入layout2，唔係就會變左佢地係同輩

                if(isRestoring) return@doOnTextChanged

fun saveHost(){
//        val noteContent = note.content.clone() as ArrayList<ContentClass>     //.clone()復製新的一個東西
//        val noteContent = ArrayList(note.content)    //呢個都係複製一份嘅意思，但係佢呢個係淺層複製，淨係可以複製你現時呢一層，再落下嗰啲複製唔到
        val noteContent = note.content.map { it.deepCopy() } as ArrayList<ContentClass>
        if(hostNumber != 0){
            Data.hostList.removeAll(Data.hostList.takeLast(hostNumber))    //棧的結構：先進後出，後入嚟嘅佢會先出去，(出去都可以被作唔要了)所以當佢喺有上一步嘅情況下改變左內容，即係佢之前啲野唔要了，就係睇佢返回左幾多就唔要幾多。或者可以咁樣理解，佢返回上一步即係將啲野移赴左，如果佢再加其他野咁就有一個棧又入左嚟，咁之前啲就唔要，佢上一步就會係加入之的一步。所以都係要deleted中間嘅所有野
            hostNumber = 0
        }
        val noteNow = NoteClass(note.title, LocalDateTime.now().toString(),noteContent)
        Data.hostList.add(noteNow)
    }

        
deleteImage.setOnClickListener {
    var list = nowId!!.split("-")
    if(list.isNullOrEmpty()){
        list = listOf(nowId) as List<String>
    }
    if(list[0] != "") {
        if (list.size > 1) {
            val noteObject = note.content[list[0].toInt()]
            removeObject(noteObject, list.takeLast(list.size - 1))   
        } else {
            note.content.removeAt(list[0].toInt())    //很容易理解，都係揾返佢喺邊跟住remove左就可以
            saveHost()
            load()
        }
    }
    nowId = ""
}
fun removeObject(noteObject: ContentClass, list: List<String>){
    if(list.isNotEmpty()){
        if(list.size > 1){
            removeObject(noteObject.children!![list[0].toInt()], list.takeLast(list.size-1))
        }
        else{
            noteObject.children!!.removeAt(list[0].toInt())
            saveHost()
            load()
        }
    }
}
