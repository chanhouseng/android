}

val today = Data.todayRecord
var cash = 0
for(result in today.stops!!){
    cash += result.cashInAmount    //計total cash
}
view.findViewById<TextView>(R.id.textView9).text = "$${cash}"
var numberKm : Double = 0.0
var i = 0
if(today.stops!!.size < 2){
    numberKm = 0.0
}
else{
    while (i < today.stops!!.size-1){    //因為佢要用到n同n+1，所以用for做唔到
        val x = (today.stops!![i].latitude.minus(today.stops!![i+1].latitude))*(today.stops!![i].latitude.minus(today.stops!![i+1].latitude))
        val y = (today.stops!![i].longitude.minus(today.stops!![i+1].longitude))*(today.stops!![i].longitude.minus(today.stops!![i+1].longitude))
        val xy = Math.sqrt((x+y).toDouble())    //計距離
        numberKm += xy
        i++
    }
}
view.findViewById<TextView>(R.id.textView10).text = "${String.format("%.2f", numberKm)}km"
view.findViewById<TextView>(R.id.textView11).text = "${today.stops!!.size}"
view.findViewById<Button>(R.id.button7).setOnClickListener {
    today.status = "completed"
    today.totalCashIn = cash
    today.totalDistanceKm = String.format("%.2f", numberKm).toFloat()
    activity.replaceFragment(dailyRunFragment())
    activity.save()
}
