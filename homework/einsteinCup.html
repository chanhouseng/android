var startX = 0f
var startY = 0f

scrollView.setOnTouchListener { v, event ->
    when(event.action){
        MotionEvent.ACTION_DOWN -> {
            startX = event.x
            startY = event.y
        }

        MotionEvent.ACTION_UP -> {
            val moveX = event.x
            val moveY = event.y

            val deltaX = moveX - startX
            val deltaY = moveY - startY

            if(Math.abs(deltaX) > Math.abs(deltaY)){
                if(deltaX < 0){
                    if(number < linearLayout.size-1){
                        number ++ 
                        var table = linearLayout.getChildAt(number) as Table     //一個ticket物件係一個table，作用係可以攞返view的顏色去控制bottomNavigationView的顏色
                        val color = (table.view.background as ColorDrawable).color
                        setColor(color)
                        val valueAnimator = ValueAnimator.ofInt(linearLayout.getChildAt(number-1).left-130, linearLayout.getChildAt(number).left-130)
                        valueAnimator.duration = 170
                        valueAnimator.interpolator = LinearInterpolator()
                        valueAnimator.addUpdateListener {
                            scrollView.scrollX = it.animatedValue as Int
                        }
                        valueAnimator.start()
                    }
                }
                else{
                    if(number > 0){
                        number --
                        var table = linearLayout.getChildAt(number) as Table
                        val color = (table.view.background as ColorDrawable).color
                        setColor(color)
                        val valueAnimator = ValueAnimator.ofInt(linearLayout.getChildAt(number+1).left-130, linearLayout.getChildAt(number).left-130)
                        valueAnimator.duration = 170
                        valueAnimator.interpolator = LinearInterpolator()
                        valueAnimator.addUpdateListener {
                            scrollView.scrollX = it.animatedValue as Int
                        }
                        valueAnimator.start()
                    }
                }
            }
        }
    }
    true
}
