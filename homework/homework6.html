class MainActivity : AppCompatActivity() {
    private lateinit var about: TextView
    private lateinit var statistic: TextView
    private lateinit var photos: TextView
    private lateinit var skills: TextView
    private fun replaceFragment(fragment: Fragment){
        val fragmentManager = supportFragmentManager
        val transaction = fragmentManager.beginTransaction()
        transaction.replace(R.id.frameLayout, fragment)
        transaction.commit()
    }
    var number = 0

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContentView(R.layout.activity_main)
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
            val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
            insets
        }
        val view : View = findViewById(R.id.view)
        val textView: TextView = findViewById(R.id.textView)
        val imageView: ImageView = findViewById(R.id.imageView)
        val map : ImageView = findViewById(R.id.imageView2)
        val down: ImageView = findViewById(R.id.imageView3)
        val i2Anim = TranslateAnimation(0f, 0f, 0f, 200f)
        i2Anim.duration = 2000
        i2Anim.fillAfter = true
        val alphaAnimation = AlphaAnimation(0f, 1f)
        alphaAnimation.duration = 1500
        alphaAnimation.fillAfter = true
        val downAnim = TranslateAnimation(0f, 0f, 0f, 70f)
        downAnim.duration = 1500
        downAnim.repeatCount = -1    // 無限重複
        downAnim.repeatMode = Animation.REVERSE    // 來回反向播放
        var startX = 0f
        var startY = 0f
        val home : ConstraintLayout = findViewById(R.id.home)
        Thread{
            Thread.sleep(1000)    //用Thread來控制動畫先後
            runOnUiThread {
                imageView.startAnimation(i2Anim)
            }
            Thread.sleep(1000)
            runOnUiThread {
                map.startAnimation(alphaAnimation)
                textView.startAnimation(alphaAnimation)
                down.startAnimation(alphaAnimation)
                down.startAnimation(downAnim)
                view.startAnimation(alphaAnimation)
                home.setOnTouchListener { _, event ->
                    when(event.action){
                        MotionEvent.ACTION_DOWN -> {
                            startX = event.x
                            startY = event.y
                        }

                        MotionEvent.ACTION_UP -> {
                            val moveX = event.x
                            val moveY = event.y

                            val delutaX = moveX - startX
                            val delutaY = moveY - startY

                            if(Math.abs(delutaY) > Math.abs(delutaX)){
                                if(delutaY < 0){
                                    val translateAnimationUp = TranslateAnimation(0f, 0f, 0f, -3000f)
                                    translateAnimationUp.duration = 900
                                    translateAnimationUp.fillAfter = true
                                    home.startAnimation(translateAnimationUp)
                                    home.visibility = View.GONE
                                    home.isEnabled = false    //讓 home 這個元件無法被點擊
                                }
                            }
                        }
                    }
                    true
                }
            }

        }.start()

        val textView4: TextView = findViewById(R.id.textView4)
        about  = findViewById(R.id.about)
        statistic = findViewById(R.id.statistic)
        photos = findViewById(R.id.photos)
        skills = findViewById(R.id.skills)
        replaceFragment(aboutFragment())
        about.setOnClickListener {
            number = 1
            replaceFragment(aboutFragment())
            textBackground()
            textView4.text = "About"
        }
        statistic.setOnClickListener {
            number = 2
            textBackground()
            replaceFragment(statisticFragment())
            textView4.text = "Statistic"
        }
        photos.setOnClickListener {
            number = 3
            replaceFragment(photosFragment())
            textBackground()
            textView4.text = "Photos"
        }
        skills.setOnClickListener {
            number = 4
            textBackground()
            replaceFragment(skillsFragment())
            textView4.text = "Skills"
        }
    }

    private fun textBackground(){
        about.setBackgroundResource(0)
        statistic.setBackgroundResource(0)
        photos.setBackgroundResource(0)
        skills.setBackgroundResource(0)
        when(number){
            1 -> about.setBackgroundResource(R.drawable.text_back)
            2 -> statistic.setBackgroundResource(R.drawable.text_back)
            3 -> photos.setBackgroundResource(R.drawable.text_back)
            4 -> skills.setBackgroundResource(R.drawable.text_back)
        }
    }
}

        next = view.findViewById(R.id.imageView21)
       scrollView = view.findViewById(R.id.scrollView)
        bottomButton = view.findViewById(R.id.bottomButton)
        linearLayout = view.findViewById(R.id.linearLayout)
        before.setOnClickListener {
            if(imageNumber != 0){
                imageNumber -= 1
                alphaButton(1)    //所有都是已經做完了+-處理
                buttonColor()
                alphaFrame(1)
                animFun(1)
            }
        }
        before.alpha = 0.3f

        next.setOnClickListener {
            if(imageNumber != 7){
                imageNumber += 1
                alphaButton(-1)
                buttonColor()
                alphaFrame(-1)
                animFun(-1)
            }
        }
        var startX = 0f
        var startY = 0f
        scrollView.setOnTouchListener { _, event ->
            when(event.action){
                MotionEvent.ACTION_DOWN -> {
                    startX = event.x
                    startY = event.y
                }

                MotionEvent.ACTION_UP -> {
                    val moveX = event.x
                    val moveY = event.y

                    val deltaX = moveX - startX
                    val deltaY = moveY - startY

                    if(Math.abs(deltaX) > Math.abs(deltaY)){
                        if(deltaX > 0){
                            if(imageNumber != 0){
                                imageNumber -= 1
                                animFun(1)
                                alphaButton(1)
                                buttonColor()
                                alphaFrame(1)
                            }
                        }
                        else{
                            if(imageNumber != 7){
                                imageNumber += 1
                                animFun(-1)
                                alphaButton(-1)
                                buttonColor()
                                alphaFrame(-1)
                            }
                        }
                    }
                }

            }
            true
        }
        var i = 2
        while (i<=8){
            bottomButton.getChildAt(i).alpha = 0.3f
            i++
        }
        return  view
    }
    private lateinit var linearLayout: LinearLayout
    fun alphaButton(number: Int){      正為向右滑，負為向左滑。(imageNumber已經是現時要展示的相片，右滑即右邊的要移走，所以+1)
        val targetLast = bottomButton.getChildAt(imageNumber + 1 + number)    //左右兩個按鈕也在這個linearLayout中，所以要+1
        targetLast.alpha = 0.3f
        val targetNow = bottomButton.getChildAt(imageNumber + 1)
        targetNow.alpha = 1f
    }
    fun alphaFrame(number : Int){
        val targetFrame = linearLayout.getChildAt(imageNumber + number)
        targetFrame.alpha = 0.3f
        val targetNow = linearLayout.getChildAt(imageNumber)
        targetNow.alpha = 1f
    }
    private lateinit var  bottomButton: LinearLayout
    fun buttonColor(){
        if(imageNumber == 0){
            before.alpha = 0.3f
        }
        else{
            before.alpha = 1f
        }
        if(imageNumber == 7){
            next.alpha = 0.3f
        }
        else{
            next.alpha = 1f
        }
    }
    private lateinit var next: ImageView
    private lateinit var before: ImageView
    fun animFun(number: Int){
        val targetView = linearLayout.getChildAt(imageNumber)    //移後現在我的相片
        val targetLast = linearLayout.getChildAt(imageNumber + number)    //向右是1，向左是-1。正負表示前向
        val valueAnimator = ValueAnimator.ofInt(targetLast.left - 240, targetView.left - 240)    //由之間的左邊移後後來的左邊
        valueAnimator.duration = 500
        var value = 0
        valueAnimator.addUpdateListener {
            value = it.animatedValue as Int
            scrollView.scrollX = value
        }
        valueAnimator.start()
    }
    private lateinit var scrollView: HorizontalScrollView
}
