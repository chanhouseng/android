        imageView.setImageResource(drawable)

        val params = imageView.layoutParams
        params.width = (40*resources.displayMetrics.density).toInt()
        params.height = (40*resources.displayMetrics.density).toInt()
        val animator = ValueAnimator.ofFloat((resources.displayMetrics.widthPixels).toFloat(),0f)
        animator.duration = 3000
        animator.interpolator = LinearInterpolator()
        animator.addUpdateListener {
            imageView.translationX = it.animatedValue as Float
            if(isCatch(imageView)){
                catchImage(imageView)
                animatorList.remove(animator)
                playLayout.removeView(imageView)
                animator.cancel()
            }
        }

            override fun onAnimationEnd(animation: Animator) {
                super.onAnimationEnd(animation)
                animatorList.remove(animator)
                playLayout.removeView(imageView)
            }
        })
        animatorList.add(animator)
        animator.start()
        i++
        textView4.text = "${i}s"    //計時
    }

}

var statX = 0f
var statY = 0f

findViewById<FrameLayout>(R.id.main).setOnTouchListener { v, event ->
    when(event.action){
        MotionEvent.ACTION_DOWN -> {
            statY = event.y
            statX = event.x
            val countDownTimer2 = object : CountDownTimer(1000, 1000){    //長按按落去就已經始計時，反正點擊會即刻取消
                override fun onFinish() {
                    if(!isGood && countCoin > 1){
                        findViewById<TextView>(R.id.textView16).visibility = View.VISIBLE
                        countCoin --
                        textView3.text = countCoin.toString()
                        start()
                    }
                }
                override fun onTick(millisUntilFinished: Long) {
                    isGood = false     //判斷是否為無敵狀態

                }
            }
            countDownTimer2.start()
        }
        MotionEvent.ACTION_UP -> {
            isGood = true
            findViewById<TextView>(R.id.textView16).visibility = View.GONE
            val moveX = event.x
            val moveY = event.y
            val deltaX = moveX - statX
            val deltaY = moveY - statY
            if(Math.abs(deltaX) > Math.abs(deltaY)){
                if(deltaY > 0){    //滑動的效果
                    AlertDialog.Builder(this)
                        .setTitle("he game is in progress. Are you sure to quit?")
                        .setNegativeButton("No"){_, _ ->
                            giveStart()
                        }
                        .setPositiveButton("Yes"){_, _ ->
                            val intent = Intent(this, MainActivity::class.java)
                            startActivity(intent)
                        }
                        .show()
                    givePause()
                }
            }
            if(deltaX == 0f && deltaY == 0f){     //點擊的效果
                if(!isTrue){     //判斷是否正在跳躍
                    isTrue = true 
                    val media = MediaPlayer.create(this@MainActivity2, R.raw.jump)
                    media.start()
                    val animator = ValueAnimator.ofFloat(0f, -450f)
                    animator.duration = 800
                    animator.addUpdateListener {
                        imageView5.translationY = it.animatedValue as Float    //唔可以用x，用x佢個Rect框冇變到
                    }
                    animator.addListener(object : AnimatorListenerAdapter(){
                        override fun onAnimationEnd(animation: Animator) {
                            super.onAnimationEnd(animation)
                            val animator2 = ValueAnimator.ofFloat(-450f, 0f)
                            animator2.duration = 1000
                            animator2.addUpdateListener {
                                imageView5.translationY = it.animatedValue as Float
                            }
                            animator2.addListener(object : AnimatorListenerAdapter(){
                                override fun onAnimationEnd(animation: Animator) {
                                    super.onAnimationEnd(animation)
                                    isTrue = false
                                }
                            })
                            animator2.start()
                        }
                    })

                    animator.start()
                }
            }
        }
    }
    true
}
        
fun giveOver(){
    var id = if(Data.rankingArray.size == 0) 0 else Data.rankingArray.last().id + 1

    save()

    val alertDialog = AlertDialog.Builder(this).setView(layout).show()

        val intent = Intent(this, MainActivity4::class.java)
        intent.putExtra("id",id )
        startActivity(intent)
    }

    val play = MediaPlayer.create(this, R.raw.game_over)

}
