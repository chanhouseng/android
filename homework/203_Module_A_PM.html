        alertDialog.dismiss()

    var Ki2 = i

                        }
                        var scaleFactor = 1f
                        val scaleDetector = ScaleGestureDetector(this@photos_MainActivity2, object :
                            ScaleGestureDetector.SimpleOnScaleGestureListener(){
                            override fun onScale(detector: ScaleGestureDetector): Boolean {
                                val scale =  detector.scaleFactor
                                val next = scale * scaleFactor
                                if(next in (1f..10f)){
                                    scaleFactor = next
                                    imageView.scaleY = next
                                    imageView.scaleX = next
                                    imageView.pivotX = detector.focusX
                                    imageView.pivotY = detector.focusY
                                }
                                return true
                            }
                            })
                        val gestureDetector = GestureDetector(this@photos_MainActivity2, object : GestureDetector.SimpleOnGestureListener(){
                            override fun onScroll(
                                e1: MotionEvent?,
                                e2: MotionEvent,
                                distanceX: Float,
                                distanceY: Float
                            ): Boolean {
                                imageView.translationX -= distanceX
                                imageView.translationY -= distanceY
                                return true
                            }
                        })

                        alert2.setOnTouchListener { v, event ->
                            scaleDetector.onTouchEvent(event)
                            gestureDetector.onTouchEvent(event)
                            true
                        }

                    }
                    layout.findViewById<TextView>(R.id.textView8).text = "Visit: ${result.visit_count}"
                    val linear = linearLayout.getChildAt(i%3) as LinearLayout
                    linear.addView(layout)
                    i++
                }
            }
        }

    })

}
