var scaleFactor = 1f
val imageView13: ImageView = view.findViewById(R.id.imageView13)
val scaleDetector = ScaleGestureDetector(activity, object : ScaleGestureDetector.SimpleOnScaleGestureListener(){
    override fun onScale(detector: ScaleGestureDetector): Boolean {
        val scale = detector.scaleFactor
        val nextScale = scaleFactor * scale
        if(nextScale in 1f..5f){
            scaleFactor = nextScale
            imageView13.scaleX = scaleFactor
            imageView13.scaleY = scaleFactor
            imageView13.pivotY = detector.focusY
            imageView13.pivotX = detector.focusX
        }
        return true
    }
})

val gestureDetector = GestureDetector(activity, object : GestureDetector.SimpleOnGestureListener(){
    override fun onScroll(
        e1: MotionEvent?,
        e2: MotionEvent,
        distanceX: Float,
        distanceY: Float
    ): Boolean {
        imageView13.translationX -= distanceX
        imageView13.translationY -= distanceY
        return true
    }
})

main.setOnTouchListener { v, event ->
    scaleDetector.onTouchEvent(event)
    gestureDetector.onTouchEvent(event)
    true
}

drawButton.setOnClickListener {
    view.findViewById<LinearLayout>(R.id.l1).setBackgroundColor((Color.parseColor("#92bcab")))
    val paint = Paint().apply {
        color = Color.RED
        style = Paint.Style.STROKE
        strokeWidth = 30f
        isAntiAlias = true
        strokeJoin = Paint.Join.ROUND
        strokeCap = Paint.Cap.ROUND
    }
    val path = Path()
    var lastX = 0f
    var lastY = 0f
    val width = resources.displayMetrics.widthPixels
    val height = resources.displayMetrics.heightPixels
    originalBitmap = BitmapFactory.decodeResource(resources, R.drawable.map_paris_city)
    mutableBitmap = Bitmap.createScaledBitmap(originalBitmap, width, height, true)
    canvas = Canvas(mutableBitmap)
    imageView13.setImageBitmap(mutableBitmap)

    imageView13.setOnTouchListener { v, event ->
        val x = event.x
        val y = event.y

        when(event.action){
            MotionEvent.ACTION_DOWN -> {path.moveTo(x, y)
                lastX = x
                lastY = y
                true
            }

            MotionEvent.ACTION_MOVE -> {
                path.quadTo(lastX, lastY, (lastX + x) / 2, (lastY + y) / 2)
                lastX = x
                lastY = y
                canvas.drawPath(path, paint)
                imageView13.invalidate()
                true
            }

            MotionEvent.ACTION_UP -> {
                path.lineTo(x, y)
                canvas.drawPath(path, paint)
                path.reset()
                imageView13.invalidate()
                true
            }

            else -> false
        }

    }
}

fun saveLayoutToGallery(view: View) {
    try {
        val bitmap = Bitmap.createBitmap(
            view.width,
            view.height,
            Bitmap.Config.ARGB_8888
        )

        val canvas = Canvas(bitmap)
        view.draw(canvas)

        saveImageToGallery(bitmap)
    } catch (e: Exception) {
        Toast.makeText(activity, "保存失敗: ${e.message}", Toast.LENGTH_SHORT).show()
    }
}

fun saveImageToGallery(bitmap: Bitmap) {
    try {
        val fileName = "drawable_${System.currentTimeMillis()}.png"
        val contentValues = ContentValues().apply {
            put(MediaStore.Images.Media.DISPLAY_NAME, fileName)
            put(MediaStore.Images.Media.MIME_TYPE, "image/png")
            put(MediaStore.Images.Media.RELATIVE_PATH, Environment.DIRECTORY_PICTURES)
        }

        val uri = requireActivity().contentResolver?.insert(
            MediaStore.Images.Media.EXTERNAL_CONTENT_URI,
            contentValues
        )
        uri?.let {
            activity?.contentResolver?.openOutputStream(it)?.use { stream ->
                bitmap.compress(Bitmap.CompressFormat.PNG, 100, stream)
            }
            true
        } ?: false
        Toast.makeText(activity, "Save to gallery", Toast.LENGTH_SHORT).show()

    } catch (e: Exception) {
        Toast.makeText(activity, e.message.toString(), Toast.LENGTH_SHORT).show()
    }
}
