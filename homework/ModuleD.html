fun showProduct(number:Int){
    showTotal()
    nowPage = number
    linearLayout.removeAllViews()
    if(number != 8){
        nothing.visibility = View.GONE
        val results = Data.foodMenuArray.filter { it.type_id == number.toString() }
        val allNumberObject2 = Data.foodMenuArray.filter { it.number > 0 }
        var allNumber2 = 0
        for(number in allNumberObject2){
            allNumber2 += number.number
        }
        if(allNumber2 == 0) carNumberText.visibility = View.INVISIBLE else {
            t10Number.text = allNumber2.toString()
            carNumberText.visibility = View.VISIBLE
        }
        for(result in results){
            val layout = layoutInflater.inflate(R.layout.car_pro , null)
            layout.findViewById<TextView>(R.id.textView3).text = result.name
            layout.findViewById<TextView>(R.id.textView2).text = result.remark
            layout.findViewById<ImageView>(R.id.imageView).setImageResource(resources.getIdentifier(result.image_url.replace(".png", "").replace(".jpeg" , "").replace(".jpg", ""), "drawable", packageName))
            layout.findViewById<TextView>(R.id.textView4).text = "$${result.price}"
            val addButton = layout.findViewById<LinearLayout>(R.id.addButton)
            val removeButton = layout.findViewById<LinearLayout>(R.id.removeButton)
            val proNumber = layout.findViewById<TextView>(R.id.textView5)
            if(result.number > 0){
                proNumber.visibility = View.VISIBLE
                removeButton.visibility = View.VISIBLE
                proNumber.text = result.number.toString()
            }
            addButton.setOnClickListener {
                result.number++
                if(result.number == 1){
                    removeButton.visibility = View.VISIBLE
                    proNumber.visibility = View.VISIBLE
                    carNumberText.visibility = View.VISIBLE
                }
                proNumber.text = result.number.toString()
                val allNumberObject = Data.foodMenuArray.filter { it.number > 0 }
                var allNumber = 0
                for(number in allNumberObject){
                    allNumber += number.number
                }
                t10Number.text = allNumber.toString()
                showTotal()

            }
            removeButton.setOnClickListener {
                if(result.number >= 1){
                    result.number--
                    if(result.number == 0){
                        removeButton.visibility = View.INVISIBLE
                        proNumber.visibility = View.INVISIBLE
                    }
                    proNumber.text = result.number.toString()
                    val allNumberObject = Data.foodMenuArray.filter { it.number > 0 }
                    var allNumber = 0
                    for(number in allNumberObject){
                        allNumber += number.number
                    }
                    if(allNumber == 0) carNumberText.visibility = View.INVISIBLE else t10Number.text = allNumber.toString()
                    showTotal()
                }
            }
            linearLayout.addView(layout)
        }
    }
    else{
        nothing.visibility = View.VISIBLE
    }
}
