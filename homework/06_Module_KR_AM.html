var lastX = 0f
var lastY = 0f
imageView.setOnTouchListener { v, event ->
    when(event.action){
        MotionEvent.ACTION_MOVE -> {
            lastX = event.x
            lastY = event.y
        }

        MotionEvent.ACTION_UP -> {
            val moveX = event.x
            val moveY = event.y

            val defaltX = moveX - lastX
            val defaltY = moveY - lastY

            if(Math.abs(defaltX) > Math.abs(defaltY)){
                if(defaltX > 0){    //邏輯同上
