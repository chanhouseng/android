val name: EditText = findViewById<EditText>(R.id.editTextText)
findViewById<TextView>(R.id.textView5).setOnClickListener {

    if(name.text.toString().length == 0){
        Toast.makeText(this, "Full name should not be empty", Toast.LENGTH_SHORT).show()
    }
    else if(name.text.toString().firstOrNull()?.isDigit() == true){
        //.firstOrNull()取出字串的第一個字元。如果字串為空（""），則返回 null
        //isDigit() 會檢查是否為 '0' ~ '9'。
        Toast.makeText(this, "Full name should not start with number", Toast.LENGTH_SHORT).show()
    }
    else if(name.text.toString().length < 3){
        Toast.makeText(this, "Full name should not be less than 3 characters", Toast.LENGTH_SHORT).show()
    }
    else{
        val intent = Intent(this, MainActivity4::class.java)
        startActivity(intent)
        val editor = getSharedPreferences("data", MODE_PRIVATE).edit()
        editor.putString("name", name.text.toString())
        editor.apply()
    }
}

            startActivity(intent)
        }
        val editTextText: EditText = findViewById(R.id.editTextText)
        val timeLimit = LocalDate.of(2010, 1, 1)
        val textView31: TextView = findViewById(R.id.textView31)
        editTextText.doOnTextChanged { text, start, before, count ->         //每次輸入都進行
            val textShow = when(text.toString().length){
                0,1,2 -> "${text.toString().padEnd(2,'D')} / MM / YYYY"
                3,4 -> "${text.toString().take(2)} / ${text.toString().substring(2, text.toString().length).padEnd(2,'M')} / YYYY"
                else -> "${text.toString().take(2)} / ${text.toString().substring(2,4)} / ${text.toString().substring(4, text.toString().length).padEnd(4,'Y')}"
            }
            textView31.text = textShow.toString()
        }
        findViewById<TextView>(R.id.textView5).setOnClickListener {
            val text = editTextText.text.toString()

            val time : LocalDate? = try {
                LocalDate.of(text.substring(4, 8).toInt(),text.substring(2, 4).toInt(),text.substring(0,2).toInt())
            }catch (e: Exception){
                null
            }
            if(time == null){
                Toast.makeText(this, "please input a correct time formatter",Toast.LENGTH_SHORT).show()
            }
            else if(time.isAfter(timeLimit)){
                Toast.makeText(this, " Birth date should be a valid date and before 2010", Toast.LENGTH_SHORT).show()
            }
            else{
                val intent = Intent(this, MainActivity5::class.java)
                startActivity(intent)
            }
        }
    }
}

class seachFragment : Fragment() {
    private lateinit var linearLayout: LinearLayout
    private lateinit var search: EditText
    private var searchId = arrayListOf<Int>()
    private var searchId2 = listOf<Int>()
    private var searchHandler = Handler(Looper.getMainLooper())   //建立排程器
    override fun onCreateView(
        inflater: LayoutInflater, container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        // Inflate the layout for this fragment
        val view =  inflater.inflate(R.layout.fragment_seach, container, false)
        val activity = requireActivity() as MainActivity8
        linearLayout = view.findViewById(R.id.linearLayout)
        search = view.findViewById(R.id.editTextText3)
        val results = Gson().fromJson(Data.universityText, Array<universityClass>::class.java)
        val results2 = Gson().fromJson(Data.studyFields, Array<studyClass>::class.java)
        search.doOnTextChanged { text, start, before, count ->
            linearLayout.removeAllViews()
            searchHandler.removeCallbacksAndMessages(null)
            if(text.toString().length > 3){
                searchHandler.postDelayed({    //三秒後才做的東西
                    searchId.clear()
                    linearLayout.removeAllViews()
                    for(result in results){
                        if(result.university_name.contains(text.toString())){
                            searchId.add(result.university_id)     //因為佢要按標題、內文等等去搜，所以合適先記ID
                        }
                    }
                    for(result in results){
                        if(result.university_desc.contains(text.toString())){
                            searchId.add(result.university_id)
                        }
                    }
                    for(result in results2){
                        for(fields in result.fields_of_study){
                            if(fields.name.contains(text.toString())){
                                searchId.add(result.university_id)
                            }}
                    }
                    searchId2 = searchId.distinct()    //必然有重覆項，就要刪去
                    for(searchId in searchId2) { 
                        val resultUniversity = results.filter { it.university_id == searchId }    //最後按ID去顯示學校
                        val studyResult = results2.filter { it.university_id == searchId }
                        for (result in resultUniversity) {
                            for (study in studyResult) {
                                val layout = layoutInflater.inflate(R.layout.search_pro, null)
                                layout.findViewById<TextView>(R.id.textView25).text =
                                    result.university_name
                                layout.findViewById<TextView>(R.id.textView26).text =
                                    result.university_location
                                layout.findViewById<TextView>(R.id.textView27).text =
                                    result.university_desc
                                val fields = study.fields_of_study.joinToString(", ") { it.name }
                                layout.findViewById<TextView>(R.id.textView28).text =
                                    "Fields: ${fields}"
                                layout.setOnClickListener {
                                    val intent = Intent(activity, MainActivity9::class.java)
                                    intent.putExtra("name", result.university_name)
                                    intent.putExtra("desc", result.university_desc)
                                    intent.putExtra("photo", result.university_photo)
                                    intent.putExtra("website", result.university_website)
                                    intent.putExtra("id", result.university_id)
                                    intent.putExtra("location", result.university_location)
                                    startActivity(intent)
                                }
                                linearLayout.addView(layout)
                            }
                        }
                    }
                }, 3000)
            }
        }
        return  view
    }
}
