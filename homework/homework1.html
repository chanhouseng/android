class MainActivity3 : AppCompatActivity() {
    private lateinit var oval1: CardView
    private lateinit var oval2: CardView
    private lateinit var oval3: CardView
    private lateinit var linearLayout1: LinearLayout
    private lateinit var linearLayout2: LinearLayout
    private lateinit var linearLayout3: LinearLayout
    val transitionArray = arrayOf(0, 450, 900)
    val colorName = arrayOf("YELLOW", "GREEN", "PURPLE", "RED", "BLUE", "ORANGE")
    val colorRBG = arrayOf("#FFC700", "#00A991", "#C00DCF", "#CF0505", "#10A8FE", "#FFA500")
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContentView(R.layout.activity_main3)
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
            val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
            insets
        }

         -- start --
        oval1 = findViewById(R.id.oval1)
        oval2 = findViewById(R.id.oval2)
        oval3 = findViewById(R.id.oval3)
        linearLayout1 = findViewById(R.id.linearLayout1)
        linearLayout2 = findViewById(R.id.linearLayout2)
        linearLayout3 = findViewById(R.id.linearLayout3)
        val imageView1: ImageView = findViewById(R.id.imageView1)
        val imageView2: ImageView = findViewById(R.id.imageView2)
        val imageView3: ImageView = findViewById(R.id.imageView3)
        val textView : TextView = findViewById(R.id.color)
        val score: TextView = findViewById(R.id.textView8)
        var grade = 0
        val name = intent.getStringExtra("name").toString()
        val img = intent.getStringExtra("img").toString()
        val imageView12: ImageView = findViewById(R.id.imageView12)
        imageView12.setImageResource(resources.getIdentifier(img, "drawable", packageName))
        findViewById<TextView>(R.id.name).text = name

        val countDownTimer = object : CountDownTimer(2000, 2000){
            override fun onFinish() {
                val intents = Intent(this@MainActivity3, MainActivity4::class.java)
                var newId = intent.getIntExtra("id", 0)
                if(newId == 0){
                    newId = if(Data.playerList.size == 0) 1 else Data.playerList.last().id + 1
                }
                if(Data.oldUser){
                    val newOne = Data.playerList.find { it.id == newId }
                    newOne?.score = grade
                }
                else{
                    Data.playerList.add(player( newId, name, img, grade))
                }
                Toast.makeText(this@MainActivity3, newId.toString(), Toast.LENGTH_SHORT).show()
                intents.putExtra("id", newId)
                intents.putExtra("name", name)
                intents.putExtra("img", img)
                intents.putExtra("grade", grade)
                startActivity(intents)
            }

            override fun onTick(millisUntilFinished: Long) {
                val randomNumber = (0..5).random()
                val colorNew1 = colorRBG.filter { it != colorRBG[randomNumber] }
                val color1 = colorNew1.random()
                val color2 = colorNew1.filter { it != color1 }.random()
                val tranY1 = (0..2).random()
                val tranNew1 = transitionArray.filter { it != transitionArray[tranY1] }
                val tranY2 = tranNew1.random()
                val tranY3 = tranNew1.filter { it != tranY2 }.random()
                linearLayout1.setBackgroundColor(Color.parseColor(colorRBG[randomNumber]))
                linearLayout2.setBackgroundColor(Color.parseColor(color1))
                linearLayout3.setBackgroundColor(Color.parseColor(color2))
                textView.setText(colorName[randomNumber])
                oval1.translationY = transitionArray[tranY1].toFloat()
                textView.setTextColor(Color.parseColor(colorRBG[randomNumber]))
                oval2.translationY = tranY2.toFloat()
                oval3.translationY = tranY3.toFloat()
                var number2 = 0
                val countDownTimer2 = object : CountDownTimer(2000, 500){
                    override fun onFinish() {
                        imageView1.setImageResource(0)
                        cancel()
                    }

                    override fun onTick(millisUntilFinished: Long) {
                        when(number2){
                            0 -> {imageView1.setImageResource(R.drawable.button2)
                                imageView2.setImageResource(R.drawable.button2)
                                imageView3.setImageResource(R.drawable.button2)}
                            1 -> {imageView1.setImageResource(R.drawable.button3)
                                imageView2.setImageResource(R.drawable.button3)
                                imageView3.setImageResource(R.drawable.button3)}
                            2 -> {imageView1.setImageResource(R.drawable.button4)
                                imageView2.setImageResource(R.drawable.button4)
                                imageView3.setImageResource(R.drawable.button4)}
                        }
                        number2 ++
                    }

                }
                countDownTimer2.start()
                linearLayout1.setOnClickListener {
                    cancel()
                    countDownTimer2.cancel()
                    start()
                    grade++
                    score.text = "Score: $grade"
                }
                linearLayout2.setOnClickListener {
                    cancel()
                    countDownTimer2.cancel()
                    onFinish()
                }
                linearLayout3.setOnClickListener {
                    cancel()
                    countDownTimer2.cancel()
                    onFinish()
                }
            }
        }
        countDownTimer.start()
    }
}
        
class MainActivity4 : AppCompatActivity() {
    private lateinit var linearLayout: LinearLayout
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContentView(R.layout.activity_main4)
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
            val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
            insets
        }
        linearLayout = findViewById(R.id.linearLayout)
        val name = intent.getStringExtra("name")
        val imgId = resources.getIdentifier(intent.getStringExtra("img"), "drawable", packageName)
        val layoutNew = layoutInflater.inflate(R.layout.product, null)
        layoutNew.findViewById<TextView>(R.id.name).text = name
        layoutNew.findViewById<TextView>(R.id.score).text = "Score: ${intent.getIntExtra("grade", 0)}"
        layoutNew.findViewById<ImageView>(R.id.imageView).setImageResource(imgId)
        linearLayout.addView(layoutNew)
        findViewById<ImageView>(R.id.imageView12).setImageResource(imgId)
        Data.oldUser = false
        findViewById<TextView>(R.id.name).text = name
        val nowId = intent.getIntExtra("id", 0)
        val playArray = Data.playerList.filter { it.id != nowId }.sortedByDescending { it.score }
        val playList = arrayListOf<player>()
        if (playArray.size > 9){
            for(user in playArray){
                if(playList.size < 9){
                    playList.add(user)
                }
                else{
                    break
                }
            }
        }
        else{
            playList.addAll(playArray)
        }
        val button2 : Button = findViewById(R.id.button2)
        button2.setOnClickListener {
            val intent = Intent(this, MainActivity2::class.java)
            startActivity(intent)
        }
        val button1 : Button = findViewById(R.id.button1)
        button1.setOnClickListener {
            Data.oldUser = true
            val intents = Intent(this@MainActivity4, MainActivity3::class.java)
            intents.putExtra("name", name)
            intents.putExtra("img", intent.getStringExtra("img"))
            intents.putExtra("id", nowId)
            startActivity(intents)
        }
        if(playList.size != 0){
            for(play in playList){
                val layout = layoutInflater.inflate(R.layout.product, null)
                layout.findViewById<TextView>(R.id.name).text = play.name
                layout.findViewById<TextView>(R.id.score).text = "Score: ${play.score}"
                layout.setOnClickListener {
                    Data.oldUser = true
                    val intent = Intent(this@MainActivity4, MainActivity3::class.java)
                    intent.putExtra("name", play.name)
                    intent.putExtra("img", play.imgName)
                    intent.putExtra("id", play.id)
                    startActivity(intent)
                }
                layout.findViewById<ImageView>(R.id.imageView).setImageResource(resources.getIdentifier(play.imgName, "drawable", packageName))
                linearLayout.addView(layout)
            }
        }
    }
}
