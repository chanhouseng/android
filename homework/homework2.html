                }
                if(grade.Facilities != null){
                    staffFloat.text = String.format("%.1f", grade.Facilities)
                    staffProgress.progress = grade.Facilities.toInt()
                }
                if (grade.Cleanliness != null){
                    moneyFloat.text = String.format("%.1f", grade.Cleanliness)
                    moneyProgress.progress = grade.Cleanliness.toInt()

                if(grade.value_for_money != null){
