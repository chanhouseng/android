countDownTimer = object : CountDownTimer(2000, 100){
    override fun onFinish() {
        val random = (0..2).random()
        imageView.setImageResource(list[random])
        Thread{
            Thread.sleep(1000)
            runOnUiThread {
                if(random == number){
                    val intent = Intent(this@MainActivity, MainActivity2::class.java)
                    Data.myRank.add(MyRank(number, random, "Draw" ))
                    startActivity(intent)
                }
                when(random){
                    0 -> when(number){
                        1 -> {
                            Data.myRank.add(MyRank(number, random, "You win" ))
                            val intent = Intent(this@MainActivity, MainActivity3::class.java)
                            startActivity(intent)
                        }
                        2 -> {
                            Data.myRank.add(MyRank(number, random, "You lose" ))
                            val intent = Intent(this@MainActivity, MainActivity4::class.java)
                            startActivity(intent)
                        }
                    }
                    1 -> when(number){
                        0 -> {
                            Data.myRank.add(MyRank(number, random, "You lose" ))
                            val intent = Intent(this@MainActivity, MainActivity4::class.java)
                            startActivity(intent)
                        }
                        2 -> {
                            Data.myRank.add(MyRank(number, random, "You win" ))
                            val intent = Intent(this@MainActivity, MainActivity3::class.java)
                            startActivity(intent)
                        }
                    }
                    2 -> when(number){
                        0 -> {
                            Data.myRank.add(MyRank(number, random, "You win" ))
                            val intent = Intent(this@MainActivity, MainActivity3::class.java)
                            startActivity(intent)
                        }
                        1 -> {
                            Data.myRank.add(MyRank(number, random, "You lose" ))
                            val intent = Intent(this@MainActivity, MainActivity4::class.java)
                            startActivity(intent)
                        }
                    }
                }
                isStart = false
            }
        }.start()
    }

    override fun onTick(millisUntilFinished: Long) {
        imageView.setImageResource(list[i%3])     //換圖片
        i++
    }

}
        
imageView4.setOnClickListener {
    if(!isStart){
        isStart = true
        number = 2
        linearLayout.visibility = View.GONE
        imageView10.visibility = View.VISIBLE
        when(number){
            2 -> imageView10.setImageResource(R.drawable.a3)
            1 -> imageView10.setImageResource(R.drawable.a2)
            0 -> imageView10.setImageResource(R.drawable.a1)
        }
        countDownTimer.start()
    }
}
