fun loadSeat(){
    Data.seatLayout.clear()
    for(result in seatList){
        val layout = layoutInflater.inflate(R.layout.seat_pro, null)
        if(result.id != ""){
            layout.findViewById<TextView>(R.id.textView6).visibility = View.GONE
            layout.findViewById<LinearLayout>(R.id.competitorLayout).visibility = View.VISIBLE
            val competitor = Data.testData3Array.competitors.find { it.id == result.id }!!
            layout.findViewById<TextView>(R.id.textView7).text = competitor.flagEmoji
            layout.findViewById<TextView>(R.id.textView8).text = competitor.name
            if(competitor.thumbnail.contains("content://")){
                layout.findViewById<ImageView>(R.id.imageView2).setImageURI(competitor.thumbnail.toUri())
            }
            else{
                layout.findViewById<ImageView>(R.id.imageView2).setImageResource(resources.getIdentifier(competitor.thumbnail.replace(".png", "").replace("assets/",""), "drawable", requireActivity().packageName))
            }

        }
        else{
            layout.findViewById<TextView>(R.id.textView6).visibility = View.VISIBLE
            layout.findViewById<LinearLayout>(R.id.competitorLayout).visibility = View.GONE
            layout.findViewById<TextView>(R.id.textView6).text = result.seat.toString()
        }
        if(result.isLog){
            layout.findViewById<ImageView>(R.id.imageView4).visibility = View.VISIBLE
        }
        else{
            layout.findViewById<ImageView>(R.id.imageView4).visibility = View.GONE
        }
        layout.setOnClickListener {
            result.isLog = !result.isLog
            if(result.isLog){
                layout.findViewById<ImageView>(R.id.imageView4).visibility = View.VISIBLE
            }
            else{
                layout.findViewById<ImageView>(R.id.imageView4).visibility = View.GONE
            }
        }
        layout.setOnLongClickListener {
            if(result.id != ""){
                competitorCard.visibility = View.VISIBLE
                selectedCard.visibility = View.GONE
                val competitor = Data.testData3Array.competitors.find { it.id == result.id }!!
                textView4.text = competitor.name
                textView2.text = "${competitor.countryName}  ${competitor.flagEmoji}"
                if(competitor.thumbnail.contains("content://")){
                    imageView.setImageURI(competitor.thumbnail.toUri())
                }else{
                    imageView.setImageResource(resources.getIdentifier(competitor.thumbnail.replace(".png", "").replace("assets/",""), "drawable", requireActivity().packageName))
                }
                textView5.text = "Seat: ${result.seat}"
            }
            true
        }
        Data.seatLayout.add(layout)    //就係用seatLayout array去裝曬所有view，到時就顯示︳但係佢唔會有關聯性，所以每次都要call之後顯示
    }
}
        
fun load(){
    val size = Data.seatLayout.size
    l1.removeAllViews()
    l2.removeAllViews()
    l3.removeAllViews()
    when(size%2){    //分奇偶數有唔同要求
        0 -> {
            val l1Size = (size-4)/2
            println(l1Size)
            for(result in Data.seatLayout.take(l1Size).reversed()){
                l1.addView(result)
            }
            for(result in Data.seatLayout.subList(l1Size, l1Size+4)){
                l2.addView(result)
            }
            for(result in Data.seatLayout.takeLast(l1Size)){
                l3.addView(result)
            }
        }
        1 -> {
            val l1Size = (size-3)/2
            println(l1Size)
            for(result in Data.seatLayout.take(l1Size).reversed()){
                l1.addView(result, 0)
            }
            for(result in Data.seatLayout.subList(l1Size, l1Size+3)){
                l2.addView(result)
            }
            for(result in Data.seatLayout.takeLast(l1Size)){
                l3.addView(result)
            }
        }
    }
}

fun loadList(){
    if(!isCome){
        Data.seatList.clear()
        var i = 1
        while (i <= Data.testData3Array.competitors.filter { it.tradeName == "Mobile Applications Development" }.size){
            Data.seatList.add(SeatClass(i))
            i++
        }

        Data.seatList2.clear()
        var k = 1
        while (k <= Data.testData3Array.competitors.filter { it.tradeName == "Software Testing" }.size){
            Data.seatList2.add(SeatClass(k))
            k++
        }
        isCome = true
    }
    when(status){
        "Mobile Applications Development" -> seatList = Data.seatList    //所有關於seatList就會每次換，唔可以用Data.seatList
        else -> seatList = Data.seatList2    //同時也有關聯性
    }
}

fun load(){
    linearLayout.removeAllViews()
    val list = when(status){
        "All trades" -> Data.testData3Array.competitors.toList()
        else -> Data.testData3Array.competitors.filter { it.tradeName == status }
    } as List<CompetitorsClass>
    for(result in list){
        val layout = layoutInflater.inflate(R.layout.people_pro, null)
        layout.findViewById<TextView>(R.id.textView12).text = result.name
        layout.findViewById<TextView>(R.id.textView13).text = "${result.tradeName}  -  ${result.countryCode}"
        if(result.thumbnail.contains("content://")){
            layout.findViewById<ImageView>(R.id.imageView5).setImageURI(result.thumbnail.toUri())
        }else{
            layout.findViewById<ImageView>(R.id.imageView5).setImageResource(resources.getIdentifier(result.thumbnail.replace(".png", "").replace("assets/",""), "drawable", requireActivity().packageName))
        }
        layout.findViewById<TextView>(R.id.textView14).setOnClickListener {
            val intent = Intent(requireActivity(), MainActivity2::class.java)
            intent.putExtra("id", result.id)
            startActivity(intent)
        }
        linearLayout.addView(layout)
    }
}
