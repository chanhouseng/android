countDownTimer = object : CountDownTimer(currentTimeLeft, 1000){
    override fun onFinish() {
        timeUse++
        currentTimeLeft = 1000L
        showTime()     //顯示分杪
        if(!isPaused){
            startCountDownTimer()    //繼續這個countdown
        }
    }

    override fun onTick(millisUntilFinished: Long) {
        currentTimeLeft = millisUntilFinished    //暫停可以攞返餘下的時間
        valueAnimator.start()    //一秒一次毫杪的效果
    }

}
