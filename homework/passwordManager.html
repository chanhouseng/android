fun makePassword(){
    val upperString = "QWERTYUIOPASDFGHJKLMNBVCXZ"    //呢個可以喺出面build左先
    val upperList = upperString.split("") as ArrayList
    upperList.removeAt(0)     //因為冇任何字去分，所以前面都有一個""嘅值
    upperList.removeAt(upperList.size-1)
    upperList.shuffle()

    val numberString = "123456789"
    val numberList = numberString.split("") as ArrayList
    numberList.removeAt(0)
    numberList.removeAt(numberList.size-1)
    numberList.shuffle()

    val lowerString = "qwertyuiopasdfghjklmnbvcxz"
    val lowerList = lowerString.split("") as ArrayList
    lowerList.removeAt(0)
    lowerList.removeAt(lowerList.size-1)
    lowerList.shuffle()

    val symbolString = "!@#$%^&*"
    val symbolList = symbolString.split("") as ArrayList
    symbolList.removeAt(0)
    symbolList.removeAt(symbolList.size-1)
    symbolList.shuffle()

    var password = arrayListOf<String>()
    val all = arrayListOf<String>()
    if(upperSwitch.isChecked){
        password.addAll(upperList.take(1))    //必定有一個或以上的條件字符
        all.addAll(upperList)
    }
    if(lowerSwitch.isChecked){
        password.addAll(lowerList.take(1))
        all.addAll(lowerList)
    }
    if(symbolSwitch.isChecked){
        password.addAll(symbolList.take(1))
        all.addAll(symbolList)
    }
    if(numberSwitch.isChecked){
        password.addAll(numberList.take(1))
        all.addAll(numberList)
    }
    all.shuffled()
    password.addAll(all.take(len.progress-password.size))    //先攞完一個再大混亂，攞返夠數字符
    password.shuffle()
    var passwordString = ""
    var i = 0
    for(char in password){
        passwordString += char
    }
    val spannString = SpannableString(passwordString)    //設置邊種字符邊種顏色
    for(char in password){
        if(char.any { it.isDigit() }){    //檢查係咪數字等等，再去改佢所在位置的顏色
           spannString.setSpan(
               ForegroundColorSpan(Color.BLUE),
               i, i+1,
               Spannable.SPAN_EXCLUSIVE_EXCLUSIVE
           )
        }
        if(char.any { !it.isLetterOrDigit() && !it.isWhitespace() }){
            spannString.setSpan(
                ForegroundColorSpan(Color.GREEN),
                i, i+1,
                Spannable.SPAN_EXCLUSIVE_EXCLUSIVE
            )
        }
        i++
    }
    passwordText.text = spannString
}
