val value1 = ValueAnimator.ofInt(0, (resources.displayMetrics.density * 100).toInt())    //先變漸層效果
value1.duration = 600
value1.interpolator = LinearInterpolator()
value1.addUpdateListener {
    view2.layoutParams.height = it.animatedValue as Int
    view2.requestLayout()
}
value1.start()
val value2 = ValueAnimator.ofInt((444*resources.displayMetrics.density).toInt(), (90*resources.displayMetrics.density).toInt())     //再將全黑拉低，透明嗰個就會自動補位
value2.interpolator = LinearInterpolator()
value2.duration = 3000
value2.addUpdateListener {
    view3.layoutParams.height = it.animatedValue as Int
    view3.requestLayout()
}
value1.addListener(object : AnimatorListenerAdapter(){
    override fun onAnimationEnd(animation: Animator) {
        super.onAnimationEnd(animation)
        value2.start()
    }
})
        
val text1 = "Experience Art"
val text2 = "We are thrilled to invite you to join us for an extraordinary event that will immerse you in the world of art."
val linear: LinearLayout = findViewById(R.id.linear)
val textView2: TextView = findViewById(R.id.textView2)
val value4 = ValueAnimator.ofInt(0, text2.length)
value4.duration = 1800
value4.addUpdateListener {
    textView2.text = text2.take(it.animatedValue as Int)
}
val button: Button  = findViewById(R.id.button)
button.alpha = 0f
value4.addListener(object : AnimatorListenerAdapter(){
    override fun onAnimationEnd(animation: Animator) {
        super.onAnimationEnd(animation)
        button.animate().alpha(1f).translationY(0*resources.displayMetrics.density).setDuration(1000).start()
    }
})
val textView: TextView = findViewById(R.id.textView)
val value3 = ValueAnimator.ofInt(0, text1.length)
value3.duration = 700
value3.addUpdateListener {
    textView.text = text1.take(it.animatedValue as Int)
}
value3.addListener(object : AnimatorListenerAdapter(){
    override fun onAnimationEnd(animation: Animator) {
        super.onAnimationEnd(animation)
        value4.start()
    }
})
value2.addListener(object : AnimatorListenerAdapter(){
    override fun onAnimationEnd(animation: Animator) {
        super.onAnimationEnd(animation)
        linear.visibility = View.VISIBLE
        value3.start()
    }
})
button.setOnClickListener {
    val intent = Intent(this, MainActivity2::class.java)
    startActivity(intent)
}
