fun load(number: Number){
    chipGroup.clearCheck()    //清除所有的check，之後再isChecked
    linearLayout.removeAllViews()
    var list = listOf<eventClass>()
        when(number){
        1 -> {
            list = Data.eventList
            chip1.isChecked = true
        }
        2 -> {
            chip2.isChecked = true
            list = Data.eventList.filter { !it.isRead }
        }
        else -> {
            chip3.isChecked = true
            list = Data.eventList.filter { it.isRead }
        }
    }
    for(result in list){
        val layout = layoutInflater.inflate(R.layout.event_pro, null)
        layout.findViewById<TextView>(R.id.textView2).text = result.title
        layout.findViewById<TextView>(R.id.textView3).text = result.description
        layout.findViewById<ImageView>(R.id.imageView).setImageResource(resources.getIdentifier(result.image.replace(".png", "").replace(".jpeg", "").replace(".jpg", ""), "drawable", requireActivity().packageName))
        if(result.isRead){
            layout.findViewById<TextView>(R.id.textView4).text = "Read"
        }
        else{
            layout.findViewById<TextView>(R.id.textView4).text = "Unread"
        }
        layout.setOnClickListener {
            result.isRead = true
            val intent = Intent(requireActivity(), MainActivity2::class.java)
            intent.putExtra("image", result.image)
            intent.putExtra("isRead", result.isRead)
            intent.putExtra("title", result.title)
            intent.putExtra("description", result.description)
            intent.putExtra("count", result.count)
            intent.putExtra("thumbnails", result.thumbnails)
            startActivity(intent)
            result.count++
        }
        linearLayout.addView(layout)
    }
}

fun showImage(number: Int) {
    val imageArray =  intent.getStringArrayExtra("thumbnails")
    val layout = layoutInflater.inflate(R.layout.event_alert, null)    //HorizontalScrollView放喺alert以滿足滑動的要求
    val imageView = layout.findViewById<ImageView>(R.id.imageView5)
    val imageView2 = layout.findViewById<ImageView>(R.id.imageView6)
    val imageView3 = layout.findViewById<ImageView>(R.id.imageView7)
    imageView.setImageResource(resources.getIdentifier(imageArray!!.get(number).replace(".png", "").replace(".jpeg", "").replace(".jpg", ""), "drawable", packageName))
    imageView2.setImageResource(resources.getIdentifier(imageArray!!.get(if(number+1>=imageArray.size) number - 2 else number + 1).replace(".png", "").replace(".jpeg", "").replace(".jpg", ""), "drawable", packageName))    //邏輯問題，舉例可想到
    imageView3.setImageResource(resources.getIdentifier(imageArray!!.get(if(number+2>=imageArray.size) number - 1 else number + 2).replace(".png", "").replace(".jpeg", "").replace(".jpg", ""), "drawable", packageName))
    val alertDialog = AlertDialog.Builder(this).setView(layout).show()
    alertDialog.window?.setBackgroundDrawable(ColorDrawable(Color.TRANSPARENT))
}
