findViewById<Button>(R.id.button5).setOnClickListener {
    if(editTextText3.text.isNotEmpty()&&
        editTextText4.text.isNotEmpty()&&
        editTextText5.text.isNotEmpty()&&
        editTextText6.text.isNotEmpty()&&
        editTextText7.text.isNotEmpty()&&
        editTextText8.text.isNotEmpty()){
        Data.height = editTextText4.text.toString()
        Data.weight = editTextText5.text.toString()
        Data.steps = editTextText7.text.toString().toInt()
        Data.water = editTextText8.text.toString().toInt()
        val intent = Intent(this, MainActivity4::class.java)
        startActivity(intent)
    }
    else{
        Toast.makeText(this, "all input fields must be filled", Toast.LENGTH_SHORT).show()
    }
}
findViewById<EditText>(R.id.editTextText4).setOnClickListener {
    val layout = layoutInflater.inflate(R.layout.number_alert, null)
    val alertDialog = AlertDialog.Builder(this).setView(layout).show()
    layout.findViewById<TextView>(R.id.textView11).text = "Tall"
    val numberPicker1 = layout.findViewById<NumberPicker>(R.id.numberPicker1)
    val numberPicker2 = layout.findViewById<NumberPicker>(R.id.numberPicker2)
    numberPicker1.maxValue = 200
    numberPicker1.minValue = 1
    numberPicker1.minValue = 180
    numberPicker2.maxValue = 9
    numberPicker2.minValue = 5
    numberPicker2.minValue = 0
    layout.findViewById<TextView>(R.id.textView13).text = "cm"
    layout.findViewById<Button>(R.id.button7).setOnClickListener {
        alertDialog.dismiss()
    }
    layout.findViewById<Button>(R.id.button6).setOnClickListener {
        alertDialog.dismiss()
        findViewById<EditText>(R.id.editTextText4).setText("${numberPicker1.value}.${numberPicker2.value}")
    }
}

fun load(view: LinearLayout){
    imageView4.setBackgroundColor(Color.TRANSPARENT)
    imageView5.setBackgroundColor(Color.TRANSPARENT)
    view.setBackgroundColor(Color.parseColor("#18abc8"))
    when(view.id){
        R.id.imageView4 -> Data.isboy = true

    }
}
