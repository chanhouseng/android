fun load(){
    linearLayout.removeAllViews()
    for(result in Data.imageList){
        val layout = layoutInflater.inflate(R.layout.image_back, null)
        layout.findViewById<ImageView>(R.id.imageView14).setImageResource(result.image)
        layout.findViewById<TextView>(R.id.textView24).text = result.status
        layout.findViewById<ImageView>(R.id.imageView15).setOnClickListener {
            linearLayout.removeView(layout)
        }
        linearLayout.addView(layout)
    }
    for(result in Data.imageUri){    //setimageUri同普通嘅drawable唔一樣，不過都可以用try去set返張相就唔使咁
        val layout = layoutInflater.inflate(R.layout.image_back, null)
        layout.findViewById<ImageView>(R.id.imageView14).setImageURI(result?.image)
        layout.findViewById<TextView>(R.id.textView24).text = result?.status
        layout.findViewById<ImageView>(R.id.imageView15).setOnClickListener {
            linearLayout.removeView(layout)
        }
        linearLayout.addView(layout)
    }
}

override fun onResume() {    //我喺到諗緊其實佢好似有時係會實時關聯，有時又唔會。最好都係寫呢個去實時update下
    super.onResume()
    val linearLayout = view.findViewById<LinearLayout>(R.id.linearLayout)
    linearLayout.removeAllViews()
    for(result in Data.alarmsList){
        val layout = layoutInflater.inflate(R.layout.alarm_pro, null)
        layout.findViewById<TextView>(R.id.textView31).text = result.time
        val drawable = when(result.status){
            0 -> R.drawable.food_dinner_svgrepo_com
            1 -> R.drawable.water_drop_fill0_wght400_grad0_opsz24
            else  -> R.drawable.etc_svgrepo_com
        }
        layout.findViewById<ImageView>(R.id.imageView23).setImageResource(drawable)
        val switch = layout.findViewById<Switch>(R.id.switch1)
        switch.isChecked = result.isTaken
        switch.setOnCheckedChangeListener { buttonView, isChecked ->
            result.isTaken = !result.isTaken
        }
        linearLayout.addView(layout)
    }

}
