shopCart.setOnClickListener {
    l1.removeAllViews()
    l2.removeAllViews()
    bottomSheet.visibility = View.VISIBLE
    shopCart.visibility = View.GONE
    var o = 0f
    for(food in Data.orderList){
        o += (food.count?.times(food.discounted_price.replace("€", "").toFloat()))!!    //計total
    }
    textView11.text = "€${String.format("%.2f", o)}"
    var k = 0
    for(result in Data.orderList){
        val layout = layoutInflater.inflate(R.layout.shop_cart_pro, null)
        layout.findViewById<TextView>(R.id.textView7).text = result.name
        layout.findViewById<TextView>(R.id.textView10).text = result.count.toString()
        layout.findViewById<TextView>(R.id.textView8).text = result.discounted_price
        layout.findViewById<TextView>(R.id.textView9).text = result.original_price
        layout.findViewById<TextView>(R.id.textView9).paintFlags = Paint.STRIKE_THRU_TEXT_FLAG
        layout.findViewById<ImageView>(R.id.imageView4).setImageResource(resources.getIdentifier(result.image_url.lowercase().replace(" ", "_").replace(".jpg", "").replace(".png", ""), "drawable", requireActivity().packageName))
        layout.findViewById<ImageView>(R.id.imageView8).setOnClickListener {    //加數量，題目冇乜要求就最大係9
            result.count?.let { it1 ->
                if(it1 <= 8){
                    result.count = result.count!! + 1
                    layout.findViewById<TextView>(R.id.textView10).text = result.count.toString()
                    var i = 0f
                    for(food in Data.orderList){
                        i += (food.count?.times(food.discounted_price.replace("€", "").toFloat()))!!    //show total
                    }
                    textView11.text = "€${String.format("%.2f", i)}"
                }
            }
        }
        layout.findViewById<ImageView>(R.id.imageView6).setOnClickListener {    //減
            result.count?.let { it1 ->
                if(it1 >= 2){
                    result.count = result.count!! - 1
                    layout.findViewById<TextView>(R.id.textView10).text = result.count.toString()
                    var i = 0f
                    for(food in Data.orderList){
                        i += (food.count?.times(food.discounted_price.replace("€", "").toFloat()))!!
                    }
                    textView11.text = "€${String.format("%.2f", i)}"
                }
            }
        }
        when(k%2){
            0 -> l1.addView(layout)
            1 -> l2.addView(layout)
        }
        k++
    }
}
        
fun showUi(linearLayout: LinearLayout){
    menu1.visibility = View.GONE
    menu2.visibility = View.GONE
    menu3.visibility = View.GONE
    val textView = linearLayout.getChildAt(1)
    textView.visibility = View.VISIBLE
    val list = when(linearLayout.id){
        R.id.menu1Layout -> Data.mainFood
        R.id.menu2Layout -> Data.pastriesFood
        else -> Data.SoupsFood
    }
    linear1.removeAllViews()
    linearLayout2.removeAllViews()
    for(result in list){
        val imageView = layoutInflater.inflate(R.layout.image_pro, null)    //上面嗰啲相都整pro，唔係要整margin就好煩
        imageView.findViewById<ImageView>(R.id.imageView1).setImageResource(resources.getIdentifier(result.image_url.lowercase().replace(" ", "_").replace(".jpg", "").replace(".png", ""), "drawable", requireActivity().packageName))
        linear1.addView(imageView)
        val layout = layoutInflater.inflate(R.layout.food1_pro, null)
        layout.findViewById<TextView>(R.id.textView3).text = result.name
        layout.findViewById<TextView>(R.id.textView5).text = result.discounted_price
        layout.findViewById<TextView>(R.id.textView6).text = result.original_price
        layout.findViewById<TextView>(R.id.textView6).paintFlags = Paint.STRIKE_THRU_TEXT_FLAG
        layout.findViewById<ImageView>(R.id.imageView5).setImageResource(resources.getIdentifier(result.image_url.lowercase().replace(" ", "_").replace(".jpg", "").replace(".png", ""), "drawable", requireActivity().packageName))
        val number = layout.findViewById<TextView>(R.id.textView30)
        val removeImage = layout.findViewById<ImageView>(R.id.removeImage)
        val addImage = layout.findViewById<ImageView>(R.id.addImage)
        addCart()
        result.count.let {    //未暗嘅都要show返啲數
            if(it<=0){
                number.visibility = View.GONE
                removeImage.visibility = View.GONE
            } else{
                number.visibility = View.VISIBLE
                removeImage.visibility = View.VISIBLE
                number.text = result.count.toString()
            }
        }
        addImage.setOnClickListener {    //加
            result.count += 1
            result.count.let { it1 ->
                if(it1==1){
                    number.visibility = View.VISIBLE
                    removeImage.visibility = View.VISIBLE
                }
                number.text = result.count.toString()
                addCart()       //shopCard 上面嗰個數字
            }
        }
        removeImage.setOnClickListener {
            result.count.let { it1 ->
                if(it1>=1){
                    result.count -= 1
                    result.count.let { it1 ->
                        if(it1==0){
                            number.visibility = View.GONE
                            removeImage.visibility = View.GONE
                        }
                        number.text = result.count.toString()
                        addCart()
                    }
                }
            }
        }
        linearLayout2.addView(layout)

    }
}
