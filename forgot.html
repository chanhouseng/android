fun createImageUri(): Uri {    //在手機的「Pictures」資料夾中建立一個新的 JPEG 圖片檔案，並取得它的 Uri 路徑
    val contentValues = ContentValues().apply {   //其實同儲存圖片差唔多，可以理解係我要set好啲相片的資料，之後再返回一個uri以表示將佢嘅資料
        put(MediaStore.Images.Media.DISPLAY_NAME, "photo_${System.currentTimeMillis()}.jpg")
        put(MediaStore.Images.Media.MIME_TYPE, "image/jpeg")
        put(MediaStore.Images.Media.RELATIVE_PATH, Environment.DIRECTORY_PICTURES)
    }
    return contentResolver.insert(MediaStore.Images.Media.EXTERNAL_CONTENT_URI, contentValues)!!
}

val piece = findViewById<ImageView>(R.id.puzzlePiece)

        var downX = 0f
        var oldX = 0f

        piece.setOnTouchListener { v, e ->

            val parent = v.parent as View
            val maxX = parent.width - v.width
            val targetX = maxX * 0.65f

            when (e.action) {

                MotionEvent.ACTION_DOWN -> {
                    downX = e.rawX
                    oldX = v.translationX
                }

                MotionEvent.ACTION_MOVE -> {
                    v.translationX =
                        (oldX + e.rawX - downX)
                            .coerceIn(0f, maxX.toFloat())
                }

                MotionEvent.ACTION_UP -> {

                    if (kotlin.math.abs(v.translationX - targetX) < 20) {

                        v.animate()
                            .translationX(targetX)
                            .start()

                        Toast.makeText(
                            this,
                            "驗證成功",
                            Toast.LENGTH_SHORT
                        ).show()

                    } else {

                        v.animate()
                            .translationX(0f)
                            .start()

                        Toast.makeText(
                            this,
                            "驗證失敗",
                            Toast.LENGTH_SHORT
                        ).show()
                    }
                }
            }

            true
        }
