        var score = 1000
        var health = 3

class GameOverActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {

        //這裡開始

        findViewById<TextView>(R.id.macaronScoreTextView).text = (Data.macaron* 20).toString()
        findViewById<TextView>(R.id.puffCountTextView).text = "x${Data.puff}"
        findViewById<TextView>(R.id.puffScoreTextView).text = (Data.puff* 30).toString()
        findViewById<TextView>(R.id.fishboneCountTextView).text = "x${Data.fishbone}"
        findViewById<TextView>(R.id.fishboneScoreTextView).text = "-${Data.fishbone * 10}"
        findViewById<TextView>(R.id.boneCountTextView).text = "x${Data.bone}"
        findViewById<TextView>(R.id.boneScoreTextView).text = "-${Data.bone * 25}"
        findViewById<TextView>(R.id.scoreTextView).text = Data.score.toString()

    }
}

        


class MainActivity : AppCompatActivity(), SensorEventListener {    //接收手機感測器的資料

    private lateinit var sensorManager: SensorManager         //管理所有感測器
    private lateinit var accelerometer: Sensor                //加速度感測器。

    private lateinit var mainFrameLayout: FrameLayout
    private lateinit var pauseLinearLayout: LinearLayout

    private lateinit var gameFrameLayout: FrameLayout
    private lateinit var playerImageView: ImageView
    private lateinit var pauseImageView: ImageView

    private lateinit var timeTextView: TextView

    private lateinit var playImageView: ImageView
    private lateinit var moveCountDownTimer: CountDownTimer

    private lateinit var foodCountDownTimer: CountDownTimer

    private lateinit var scoreTextView: TextView
    private lateinit var heartTextView: TextView
    private var currentDirection: Direction = Direction.STAY    //記錄目前要移動的方向

    private var screenWidth = 800
    private var isGamePausing = false
 
    private val fallingFoodAnimatorList = mutableListOf<ValueAnimator>()   //儲存所有正在掉落的食物的valueAnimator

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContentView(R.layout.activity_main)
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
            val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
            insets
        }

        //這裡開始
        sensorManager = getSystemService(SENSOR_SERVICE) as SensorManager              //取得系統感測器服務與加速度計
        accelerometer = sensorManager.getDefaultSensor(Sensor.TYPE_ACCELEROMETER)!!

        mainFrameLayout = findViewById(R.id.mainFrameLayout)
        pauseLinearLayout = findViewById(R.id.pauseLinearLayout)
        timeTextView = findViewById(R.id.timeTextView)
        playImageView = findViewById(R.id.playImageView)
        gameFrameLayout = findViewById(R.id.gameFrameLayout)
        playerImageView = findViewById(R.id.playerImageView)
        pauseImageView = findViewById(R.id.pauseImageView)
        scoreTextView = findViewById(R.id.scoreTextView)
        heartTextView = findViewById(R.id.heartTextView)

        screenWidth = Resources.getSystem().displayMetrics.widthPixels     //動態取得螢幕寬度

        foodCountDownTimer = object : CountDownTimer(60000, 1000) {
            override fun onFinish() {
                start()
            }

            override fun onTick(p0: Long) {
                dropFood()
            }
        }
        foodCountDownTimer.start()


        moveCountDownTimer = object : CountDownTimer(10000, 30) {
            override fun onFinish() {
                start()
            }

            override fun onTick(p0: Long) {
                if(currentDirection == Direction.LEFT) {
                    
                    val originalLeft = playerImageView.left    // 獲得.xml一開始的左邊界位置

                    
                    val actualScreenPosition = originalLeft + playerImageView.translationX     //因為之後會有translationX 改變，所以這個是獲得實時的位置

                    if(actualScreenPosition > 0f) {
                        playerImageView.translationX = playerImageView.translationX - 10       //左移
                    } else {
                        cancel()      不要再有這個判斷
                    }
                } else if(currentDirection == Direction.RIGHT) {
                    
                    val originalRight = playerImageView.right

                    
                    val actualScreenPosition = originalRight + playerImageView.translationX

                    if(actualScreenPosition < screenWidth) {
                        playerImageView.translationX = playerImageView.translationX + 10
                    } else {
                        cancel()
                    }
                }
            }
        }


        playerImageView.setOnClickListener {
        }

        pauseImageView.setOnClickListener {
            isGamePausing = true
            gamePause()
        }

        playImageView.setOnClickListener {
            isGamePausing = false
            gameStart()
        }
    }

    private fun moveLeft() {
        if(!isGamePausing) {
            currentDirection = Direction.LEFT
            playerImageView.scaleX = 1f
            moveCountDownTimer.cancel()
            moveCountDownTimer.start()
        }
    }

    private fun moveRight() {
        if(!isGamePausing) {
            currentDirection = Direction.RIGHT 
            playerImageView.scaleX = -1f       //-1f: 水平鏡像翻轉（角色面向右邊）
            moveCountDownTimer.cancel()
            moveCountDownTimer.start()
        }
    }

    private fun dropFood() {
        val foodNameArray = arrayOf("puff", "macaron", "baguette", "fishbone", "bone")

        val foodImageView = ImageView(this)
        foodImageView.tag = foodNameArray.random()     //自訂資料，之後會用來判斷

        val drawable = when(foodImageView.tag) {
            "puff" -> R.drawable.puff
            "macaron" -> R.drawable.macaron
            "baguette" -> R.drawable.baguette
            "fishbone" -> R.drawable.fishbone
            else -> R.drawable.bone
        }
        foodImageView.setImageResource(drawable)

        gameFrameLayout.addView(foodImageView)

        val params = foodImageView.layoutParams
        params.width = 120
        params.height = 120

        foodImageView.translationX = (0..1000).random().toFloat()
        val animator = ValueAnimator.ofFloat(0f, 1500f) 
        animator.interpolator = LinearInterpolator()      //讓動畫速度均勻
        animator.duration = 4000

        animator.addUpdateListener {
            foodImageView.translationY = it.animatedValue as Float
            if(isCatching(foodImageView)) {
                catchFoodHandle(foodImageView)
                gameFrameLayout.removeView(foodImageView)
                fallingFoodAnimatorList.remove(animator)
                animator.cancel()
            }
        }

        animator.addListener(object : AnimatorListenerAdapter() {
            override fun onAnimationEnd(animation: Animator) {
                gameFrameLayout.removeView(foodImageView)
                fallingFoodAnimatorList.remove(animator)
            }
        })

        fallingFoodAnimatorList.add(animator)

        animator.start()
    }

    private fun gamePause() {
        moveCountDownTimer.cancel()
        foodCountDownTimer.cancel()

        pauseLinearLayout.visibility = View.VISIBLE
        pauseLinearLayout.bringToFront()
        mainFrameLayout.setBackgroundColor(Color.parseColor("#CC000000"))
        timeTextView.setTextColor(Color.WHITE)

        for(foodAnimator in fallingFoodAnimatorList) { 
            foodAnimator.pause()          //暫停Animator
        }
    }

    private fun gameStart() {
        moveCountDownTimer.start()
        foodCountDownTimer.start()

        pauseLinearLayout.visibility = View.GONE
        mainFrameLayout.setBackgroundColor(Color.WHITE)
        timeTextView.setTextColor(Color.BLACK)

        for (foodAnimator in fallingFoodAnimatorList) {
            foodAnimator.resume()          //暫停的那一刻繼續往下執行Animator
        }
    }

    fun getViewRect(view: View): Rect {
        val location = IntArray(2)          //建立一個長度為 2 的整數陣列，用來存放 View 的螢幕座標
        view.getLocationOnScreen(location)  //把 View 在整個螢幕上的左上角座標，存放到 location 陣列中。
        return Rect(
            location[0],
            location[1],
            location[0] + view.width,
            location[1] + view.height    //建立一個矩形（Rect）物件。左上角同右下角
        )
    }

    fun isCatching(food: View): Boolean {
        val personRect = getViewRect(playerImageView)
        val foodRect = getViewRect(food)
        return Rect.intersects(personRect, foodRect)    //檢查兩個矩形是否有重疊
    }

    fun catchFoodHandle(foodImageView: ImageView) {
        when(foodImageView.tag) {    //檢測碰到甚麼食物
            "puff" -> {
                Data.puff++
                Data.score += 30
                Data.time += 30
            }
            "baguette" -> {
                Data.baguette++
                Data.score += 10
            }
            "macaron" -> {
                Data.macaron++
                Data.score += 20
            }
            "fishbone" -> {
                Data.fishbone++
                Data.score -= 20
                Data.health -= 1
            }
            else -> {
                Data.bone++
                Data.score -= 10
            }
        }
        updateUI()      //更新生命和分數
        checkGameOver() //判斷生命是否少於0
    } 

    private fun updateUI() {
        scoreTextView.text = Data.score.toString()
        heartTextView.text = "x${Data.health}"
    }

    private fun checkGameOver() {
        if(Data.health <= 0) {
            val intent = Intent(this, GameOverActivity::class.java)
            startActivity(intent)
            gamePause()

        }
    }

    override fun onAccuracyChanged(p0: Sensor?, p1: Int) {    //處理感測器精準度變化

    }
 
    override fun onSensorChanged(event: SensorEvent) {      //感測器數值改變時
        if (event.sensor.type == Sensor.TYPE_ACCELEROMETER) {   //確認這次變化是來自加速度感測器
            val x = event.values[0]     //// 左右傾斜（-10 到 10）

            // 判斷方向（根據 X 軸數值）
             when {
                x < -3 -> moveRight()
                x > 3 -> moveLeft()
                else -> moveCountDownTimer.cancel()
            }
        }
    }

    override fun onResume() {
        super.onResume()
        sensorManager.registerListener(this, accelerometer, SensorManager.SENSOR_DELAY_UI)  //開啟感測器監聽
    }

    override fun onPause() {
        super.onPause()
        sensorManager.unregisterListener(this)  //關閉感測器監聽。
    }
}

enum class Direction {
    LEFT, STAY, RIGHT;    //建立一種有不同狀態的類型(類似變數，可用作判斷)
}
