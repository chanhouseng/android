fun load(){
    linearLayout.removeAllViews()
    for(lan in Data.languageList){
        if(lan.setList.containsAll(checkChip)){    //如果 lan.setList 這個集合「完全包含」 checkChip 裡的所有元素，就執行 { } 裡面的程式碼。
            val layout = layoutInflater.inflate(R.layout.product, null)
            layout.findViewById<TextView>(R.id.title).text = lan.title
            for(lans in lan.setList){
                val textView = TextView(this)
                textView.text = lans
                layout.findViewById<ChipGroup>(R.id.linearLayout).addView(textView)
            }
            linearLayout.addView(layout)
        }

    }
}
