linearLayout = findViewById(R.id.lienarLayout)
frameLayout = findViewById(R.id.frameLayout)
imageView = findViewById(R.id.imageView)
var startX = 0f
var startY = 0f
linearLayout.setOnTouchListener {_, e ->
    when(e.action){
        MotionEvent.ACTION_DOWN -> {
            startX = e.x
            startY = e.y
        }

        MotionEvent.ACTION_UP -> {
            val moveX = e.x
            val moveY = e.y

            val deltaX = moveX - startX
            val deltaY = moveY - startY

            if(Math.abs(deltaX) > Math.abs(deltaY)){
                if(deltaX < 0){
                    goImageView(-50f, 0f)

                }
                else{
                    goImageView(50f, 0f)
                    Toast.makeText(this@MainActivity, "y", Toast.LENGTH_SHORT).show()  //由左向右滑
                }
            }
            else{
                if(deltaY < 0){
                    goImageView(0f, -50f)

                }
                else{
                    goImageView(0f, 50f)
                    Toast.makeText(this@MainActivity, "s", Toast.LENGTH_SHORT).show()  //由上向下滑
                }
            }
        }
    }
    true
}


    fun goImageView(x: Float, y: Float){
        imageView.translationX = imageView.translationX + x
        imageView.translationY = imageView.translationY + y
    }
