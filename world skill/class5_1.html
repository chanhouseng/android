AndroidManifest.xml
<uses-permission android:name="android.permission.INTERNET"/>
android:usesCleartextTraffic="true"

val req = Request.Builder().url("http://172.18.15.142:8000/data.php").build()
OkHttpClient().newCall(req).enqueue(object : Callback{
    override fun onFailure(call: Call, e: IOException) {
        runOnUiThread {
            Toast.makeText(this@MainActivity, "please check your network", Toast.LENGTH_SHORT).show()
        }
    }

    override fun onResponse(call: Call, response: Response) {
        val resultString = response.body?.string()
        val results = Gson().fromJson(resultString, getClass::class.java)
        runOnUiThread {
            for(result in results.data){
                val layout = layoutInflater.inflate(R.layout.product, null)
                layout.findViewById<TextView>(R.id.data).text = result.date
                layout.findViewById<TextView>(R.id.job).text = result.jobactivity
                for(isCheck in result.records){
                    val imageView = ImageView(this@MainActivity)
                    if(isCheck == 1){
                        imageView.setImageResource(R.drawable.outline_check_24)
                    }
                    else if(isCheck == 0){
                        imageView.setImageResource(R.drawable.outline_award_star_24)
                    }
                    else{
                        Toast.makeText(this@MainActivity, "wrong", Toast.LENGTH_SHORT).show()
                    }
                    layout.findViewById<LinearLayout>(R.id.linearLayout).addView(imageView)
                }
                linearLayout.addView(layout)
            }
        }
    }

})
