圖片手指放大縮小、平移 

val imageMatrix = Matrix()    //所有放大、縮小、移動的操作，都記錄在此，最後再套到圖片上。
var scaleFactor = 1f      //記錄目前圖片放大的倍率
val scaleImageView: ImageView = findViewById(R.id.imageView)     
scaleImageView.scaleType = ImageView.ScaleType.MATRIX     //不要用預設的縮放方式 用『矩陣』來控制你。

// 圖像放大縮小
val scaleDetector = ScaleGestureDetector(this, object : ScaleGestureDetector.SimpleOnScaleGestureListener(){   //手指張開或合起來的監聽器
    override fun onScale(detector: ScaleGestureDetector): Boolean {     //當手指正在捏合或張開時.....
        val scale = detector.scaleFactor     //取得這一次手指移動的「縮放比例」（例如 1.05 表示稍微放大一點）。
        val nextScale = scaleFactor * scale    //計算「如果套用這次變化後，總共會變成幾倍」。
        if (nextScale in 1f..5f){      //只能從 1 倍放大到最多 5 倍，避免縮太小或放太大壞掉。
            imageMatrix.postScale(scale, scale, detector.focusX, detector.focusY) 
            //postScale -- 在目前的變形矩陣上再加上縮放的效果
            //"scale, scale" -- X方向和Y方向都放大相同的倍率（等比例縮放）
            //"detector.focusX, detector.focusY" -- 縮放的中心點（手指捏合的位置）
            scaleImageView.imageMatrix = imageMatrix    //把剛剛做好的變形套用到圖片上，畫面立刻更新。
        }
        return true    //告訴系統：「這個觸摸事件我處理完了。」
    }
})

// 圖像位移
val gestureDetector = GestureDetector(this, object : GestureDetector.SimpleOnGestureListener(){     //滑動偵測器
    override fun onScroll(     //當手指按住圖片並移動時......
        e1: MotionEvent?, 
        e2: MotionEvent,
        distanceX: Float,
        distanceY: Float
    ): Boolean {
        imageMatrix.postTranslate(-distanceX, -distanceY)    //把圖片往手指滑動的相反方向移動。
（為什麼是負號？因為手指往右滑，圖片要往左「跟著走」，才會有拖動的感覺）

        scaleImageView.imageMatrix = imageMatrix    //再次把變形套用到圖片上，讓畫面更新。
        return  true
    }
})

// 設置接觸事件 
scaleImageView.setOnTouchListener { _, event ->     當手指碰到圖片時，把觸摸事件同時傳給「捏合偵測器」和「滑動偵測器」。
    scaleDetector.onTouchEvent(event)
    gestureDetector.onTouchEvent(event)
    true
}

拖曳功能

val imageView2: ImageView = findViewById(R.id.macaron)
imageView2.setOnLongClickListener {
    val dragData = ClipData.newPlainText("drag_data", "data")    //建立一個「拖曳資料包」。告訴系統這是要被拖的東西
    val dragShadowBuilder = View.DragShadowBuilder(it)    //建立拖曳時的影子（就是你拖著移動時看到的那個半透明圖片）。
    it.startDragAndDrop(dragData, dragShadowBuilder, it, 0)      //真正開始拖曳這張圖片
    it.visibility = View.INVISIBLE    //把原本的圖片藏起來，避免畫面上同時出現兩個一樣的圖片。
    true
}

val destinationFrameLayout: FrameLayout = findViewById(R.id.frameLayout)    //找到那個可以接收拖曳的大容器（FrameLayout）。

val imageView3: ImageView = findViewById(R.id.puff)
imageView3.setOnLongClickListener {
    val dragData = ClipData.newPlainText("drag_data", "data")
    val dragShadowBuilder = View.DragShadowBuilder(it)

    it.startDragAndDrop(dragData, dragShadowBuilder, it, 0)
    it.visibility = View.INVISIBLE
    true
}

destinationFrameLayout.setOnDragListener {view, dragEvent ->    //當有東西被拖到這個容器上方時，這個監聽器就會一直監視各種拖曳事件。
    when(dragEvent.action){
        DragEvent.ACTION_DRAG_STARTED -> true    //當開始拖曳時 → 允許接收
        DragEvent.ACTION_DRAG_EXITED -> true     //當拖曳物離開容器時 → 什麼都不做。
        DragEvent.ACTION_DROP ->{      //當使用者放開手指把東西放下時
            val v = dragEvent.localState as ImageView     //取得原本被拖的那張圖片。
            val destination = view as FrameLayout     //取得目前這個容器。
            var myView = ImageView(this)
            if(v.id == R.id.macaron){     //判斷拖進來的是馬卡龍還是泡芙，然後把對應的圖片放進 myView。
                myView.setImageResource(R.drawable.macaron)
            }
            else{
                myView.setImageResource(R.drawable.puff)
            }
            if(destination.contains(v)){
                destination.removeView(v)
                myView = v     //如果原本的圖片已經在容器裡，就把它移走，然後直接使用原本的那張（避免重複）。
            }
            myView.x = dragEvent.x - 65f
            myView.y = dragEvent.y - 65f
            //把圖片放在手指放開的位置（減 65 是為了讓圖片中心對齊手指）。

            myView.layoutParams = ViewGroup.LayoutParams(150, 150)     //設定這張圖片的大小為 150×150。
            myView.setOnLongClickListener {     //給這張新放進來的圖片也加上長按可以再次拖曳的功能（讓它可以重複移動）。
                val dragData2 = ClipData.newPlainText("drag_data", "data")
                val dragShadowBuilder2 = View.DragShadowBuilder(it)
                it.startDragAndDrop(dragData2, dragShadowBuilder2, it, 0)
                it.visibility = View.INVISIBLE
                true
            }

            destination.addView(myView)

            v.visibility = View.VISIBLE    //把原本隱藏的那張圖片恢復顯示（如果需要的話）。
            true
        }

        DragEvent.ACTION_DRAG_ENDED -> true
        else -> false
        //拖曳結束時允許，其他情況不處理。
    }
    true
}


imgView2.setOnLongClickListener就係移動圖片 但拉入去要增加多一張 所以就有setOnDragListener
