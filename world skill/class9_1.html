class MainActivity2 : AppCompatActivity() {
    private lateinit var main: ConstraintLayout
    private lateinit var imageView: ImageView
    private lateinit var originalBitmap: Bitmap
    private lateinit var mutableBitmap: Bitmap
    private lateinit var canvas: Canvas
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContentView(R.layout.activity_main2)
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
            val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
            insets
        }

        main = findViewById(R.id.main)
        imageView = findViewById(R.id.imageView)

        val paint = Paint().apply{    //建立畫筆設定
            color = Color.WHITE
            style = Paint.Style.STROKE    //「只畫邊框線條」，不填滿內部。畫一個圓時，只會看到白色圓圈，圓心是空的（透明的）；如果是 FILL 就會塗滿白色。
            strokeWidth = 30f    //線條寬度設為 30 像素。
            isAntiAlias = true    //讓斜線或圓弧的邊緣不會有鋸齒狀的馬賽克。
            strokeJoin = Paint.Join.ROUND    //線條轉角和結尾都是圓滑的
            strokeCap = Paint.Cap.ROUND     //讓線條更平滑、没有鋸齒
        }

        val path = Path()    //用來記錄手指移動的軌跡
        var lastX = 0f
        var lastY = 0f

        val width = resources.displayMetrics.widthPixels
        val height = resources.displayMetrics.heightPixels

        originalBitmap = BitmapFactory.decodeResource(resources, R.drawable.croissant)    //從專案的資料夾裡讀取名為 croissant 的圖片，並轉成 Bitmap 格式。這是原始版本的圖片，程式後面會把它當作「模板」使用。
        mutableBitmap = Bitmap.createScaledBitmap(originalBitmap, width, height, true)    //把剛剛載入的 originalBitmap 放大或縮小，變成一張新的 Bitmap。這是可修改版本的 Bitmap。之後畫畫的所有線條都會畫在這張圖片上。

        canvas = Canvas(mutableBitmap)    //建立一個畫布，並把它綁定到 mutableBitmap 這張圖片上。以後都會直接畫在這張 mutableBitmap 上面。
        imageView.setImageBitmap(mutableBitmap)    //之後所有用畫筆（paint）畫的線條，都會直接留在這張 mutableBitmap 紙上，永遠改變這張紙。再把這張紙顯示在 ImageView 上，讓你看到。

        imageView.setOnTouchListener { v, event ->
            val x = event.x
            val y = event.y

            when(event.action){
                MotionEvent.ACTION_DOWN -> {
                    path.moveTo(x, y)    //把畫筆的起點放在手指位置。
                    lastX = x
                    lastY = y
                    true
                }

                MotionEvent.ACTION_MOVE -> {
                    path.quadTo(lastX, lastY, (lastX+x) / 2, (lastY + y) /2 )     //讓線條更平滑
                    lastX = x
                    lastY = y
                    canvas.drawPath(path, paint)    //把這段路徑用白色畫筆畫在紙上。
                    imageView.invalidate()    //紙變了，快點重新顯示！
                    true
                }

                MotionEvent.ACTION_UP -> { 
                    path.lineTo(x, y)     //把線條完整接到手指離開的位置。
                    canvas.drawPath(path, paint)    再一次把最終線條畫上去。
                    path.reset()    //清空這條路徑，下一筆不會跟上一筆連在一起
                    imageView.invalidate()
                    true
                }
                else -> false
            }
        }

        findViewById<Button>(R.id.button5).setOnClickListener {
            canvas.drawColor(Color.RED)
            imageView.invalidate()
            mutableBitmap = Bitmap.createScaledBitmap(originalBitmap, width, height, true)
            canvas = Canvas(mutableBitmap)
            imageView.setImageBitmap(mutableBitmap)
        }

        findViewById<Button>(R.id.button6).setOnClickListener {
            saveLayoutToGallery(main)
        }
    }

    fun saveLayoutToGallery(view: View){
        try {
            val bitmap = Bitmap.createBitmap(
                view.width,
                view.height,
                Bitmap.Config.ARGB_8888
            )

            val canvas = Canvas(bitmap)
            view.draw(canvas)

            saveImageToGallery(bitmap)
        }catch (e: Exception){
            Toast.makeText(this, "保存失敗: ${e.message}", Toast.LENGTH_SHORT).show()
        }
    }

    fun saveImageToGallery(bitmap: Bitmap){
        try {
            val fileName = "drawing_${System.currentTimeMillis()}.png"

            val contentValues = ContentValues().apply {
                put(MediaStore.Images.Media.DISPLAY_NAME, fileName)
                put(MediaStore.Images.Media.MIME_TYPE, "image/png")
                put(MediaStore.Images.Media.RELATIVE_PATH, Environment.DIRECTORY_PICTURES)
            }

            val uri = contentResolver.insert(
                MediaStore.Images.Media.EXTERNAL_CONTENT_URI,
                contentValues
            )

            uri?.let {
                contentResolver.openOutputStream(it)?.use { stream ->
                    bitmap.compress(Bitmap.CompressFormat.PNG, 100, stream)
                }
                true
            }?: false

            Toast.makeText(this, "Save to gallery", Toast.LENGTH_SHORT).show()
        }catch (e: Exception){
            Toast.makeText(this, e.message.toString(), Toast.LENGTH_SHORT).show()
            false
        }
    }
}
