val seekBar: SeekBar = findViewById(R.id.seekBar)
val seekText : TextView = findViewById(R.id.textView)

    override fun onProgressChanged(
        seekBar: SeekBar?,
        progress: Int,
        fromUser: Boolean
    ) {
        seekText.setTextSize(progress.toFloat())
    }

    override fun onStartTrackingTouch(seekBar: SeekBar?) {
    }

    override fun onStopTrackingTouch(seekBar: SeekBar?) {
    }

})

findViewById<TextView>(R.id.textView6).setOnClickListener {
    val popupMenu = PopupMenu(this, it)
    popupMenu.menuInflater.inflate(R.menu.menu, popupMenu.menu)
    popupMenu.setOnMenuItemClickListener {
        when(it.itemId){
            R.id.name -> Toast.makeText(this, "name", Toast.LENGTH_SHORT).show()
            R.id.date -> Toast.makeText(this, "date", Toast.LENGTH_SHORT).show()
        }
        true
    }
    popupMenu.show()
}

val linearLayout: LinearLayout = findViewById(R.id.linearLayout)
val valueAnimator = ValueAnimator.ofInt(3, 300)
valueAnimator.duration = 3000
valueAnimator.interpolator = DecelerateInterpolator()
var value = 0
valueAnimator.addUpdateListener {
    val value = it.animatedValue as Int
    linearLayout.layoutParams.width = value
    linearLayout.requestLayout()
}
linearLayout.setOnClickListener {
    valueAnimator.start()
}
