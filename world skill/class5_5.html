val defaultList = arrayOf(
    chipClass(1, "HK"),
    chipClass(2, "Macau"),
    chipClass(3, "China")
)
val prefs = getSharedPreferences("data", MODE_PRIVATE)
val prefsList = Gson().fromJson(prefs.getString("dataJson", Gson().toJson(defaultList).toString()),
    Array<chipClass>::class.java)
chipList.addAll(prefsList)
for(checked in prefs.getStringSet("chipChecked", emptySet<String>())?:emptySet<String>()){
    chipChecked.add(checked)
}
for(chips in chipList){
    val chip = Chip(this)
    chip.text = chips.content
    chip.isCheckable = true
    for(checked in chipChecked){
        if (chips.id.toString() == checked){
            chip.isChecked = true
        }
    }
    chip.setOnCheckedChangeListener { _, isChecked ->
        if (isChecked) chipChecked.add(chips.id.toString()) else chipChecked.remove(chips.id.toString())
        saveChip()
    }
    chip.setOnLongClickListener {
        chipGroup.removeView(chip)
        chipList.remove(chips)
        saveChip()
        true
    }
    chipGroup.addView(chip)
}
button.setOnClickListener {
    val chip = Chip(this)
    chip.text = editText.text.toString()
    val chipNew = chipClass(chipList.last().id + 1, editText.text.toString())   //先定一個class 方便之刪去
    chipList.add(chipNew)
    chip.isCheckable = true
    chip.setOnCheckedChangeListener { _, isChecked ->
        if (isChecked) chipChecked.add(chipNew.id.toString()) else chipChecked.remove(chipNew.id.toString())
        saveChip()
    }
    chip.setOnLongClickListener {
        chipGroup.removeView(chip)
        chipList.remove(chipNew)
        saveChip()
        true
    }
    chipGroup.addView(chip)
    editText.setText("")
}

    fun saveChip(){
        val editor = getSharedPreferences("data", MODE_PRIVATE).edit()
        editor.putString("dataJson", Gson().toJson(chipList))
        editor.putStringSet("chipChecked", chipChecked)
        editor.apply()
    }
