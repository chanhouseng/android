val dataButton: EditText = findViewById(R.id.dataButton)

dataButton.setOnFocusChangeListener{_, hasFocus ->
    if(hasFocus){
        dataPickerDialog.show()
        dataPickerDialog.setOnDateSetListener { _, year, month, dayOfMonth ->
            val dataString = String.format("%d-%02d-%02d", year, month+1, dayOfMonth)
            dataButton.setText(dataString)
        }
    }
}

val countDownTimer = object : CountDownTimer(3000, 1000){
    override fun onFinish() {
        number ++
        start()
    }

    override fun onTick(millisUntilFinished: Long) {
        when(number % 2){
            0 -> color.setTypeface(null, Typeface.NORMAL)
            else -> color.setTypeface(null, Typeface.BOLD)
        }
    }
}

//spinnerAdapter 的內容
class spinnerAdapter(context: Context, list: List<prodcutClass>): ArrayAdapter<prodcutClass>(context, 0, list) {  
    override fun getView(position: Int, convertView: View?, parent: ViewGroup): View {
        return initView(position, convertView, parent)
    }

    override fun getDropDownView(position: Int, convertView: View?, parent: ViewGroup): View? {
        return initView(position, convertView, parent)
    }
    private fun initView(position: Int, convertView: View?, parent: ViewGroup): View {
        val layout = LayoutInflater.from(context).inflate(R.layout.product, null)
        val item = getItem(position)
        layout.findViewById<TextView>(R.id.textView).text = item!!.name
        layout.findViewById<ImageView>(R.id.imageView).setImageResource(item!!.img)
        return layout
    }

}
