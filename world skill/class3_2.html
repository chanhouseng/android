val add : Button = findViewById(R.id.add)
val linearLayout : LinearLayout = findViewById(R.id.linearLayout)
add.setOnClickListener {
    val textView = TextView(this)
    val custom = layoutInflater.inflate(R.layout.alert, null)
    val editText = custom.findViewById<EditText>(R.id.editTextText)
    val close = custom.findViewById<ImageView>(R.id.imageView)
    val button = custom.findViewById<Button>(R.id.button)
    val alertDialog = AlertDialog.Builder(this).setView(custom).show()
    close.setOnClickListener {
        alertDialog.dismiss()
    }
    button.setOnClickListener {
        textView.text = editText.text.toString()
        alertDialog.dismiss()
        linearLayout.addView(textView, 0)  增加到linearLayout第一項
    }
    alertDialog.window?.setBackgroundDrawable(ColorDrawable(Color.TRANSPARENT))
    alertDialog.setCanceledOnTouchOutside(false)
}

val foodList = listOf("macau", "hk", "apple", "china")
val editText2 : EditText = findViewById(R.id.editTextText2)
val button2: Button = findViewById(R.id.button2)
val linearLayout2: LinearLayout = findViewById(R.id.linearLayout2)
val scrollView: ScrollView = findViewById(R.id.scrollView)
for(food in foodList){
    val layout = layoutInflater.inflate(R.layout.food, null)
    layout.findViewById<TextView>(R.id.textView).text = food
    layout.setOnLongClickListener {
        linearLayout2.removeView(layout)
        true
    }
    linearLayout2.addView(layout)
}
button2.setOnClickListener {
    val layout = layoutInflater.inflate(R.layout.food, null)
    layout.findViewById<TextView>(R.id.textView).text = editText2.text.toString()

    layout.setOnLongClickListener {
        linearLayout2.removeView(layout)
        //removeView(0, 2)刪除1,2項
        true
    }
    linearLayout2.addView(layout)
    scrollView.fullScroll(View.FOCUS_UP)
    editText2.setText("")
}
scrollView.post { //會自動最後先開始做
    scrollView.fullScroll(View.FOCUS_DOWN)
}
findViewById<Button>(R.id.up).setOnClickListener {
    scrollView.fullScroll(View.FOCUS_UP)
}
