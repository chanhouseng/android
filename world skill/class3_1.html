以layout模版自製spinner 

val foodList: List<Array> = listOf(
    Array("one", R.drawable.cupcake),
    Array("Two", R.drawable.croissant)
)

val spinner: Spinner = findViewById(R.id.spinner)
spinner.adapter = adapter(this, foodList)


Class- adapter

class adapter(context: Context, list: List<Array>): ArrayAdapter<Array>(context, 0, list) {
    override fun getView(position: Int, convertView: View?, parent: ViewGroup): View {
        return initView(position, convertView, parent)
    }

    override fun getDropDownView(position: Int, convertView: View?, parent: ViewGroup): View? {
        return initView(position, convertView, parent)
    }
    private fun initView(position: Int, convertView: View?, parent: ViewGroup): View{
        val layout = LayoutInflater.from(context).inflate(R.layout.porduct, null)
        val item = getItem(position)
        layout.findViewById<TextView>(R.id.name).text = item!!.name
        layout.findViewById<ImageView>(R.id.img).setImageResource(item!!.img)
        return layout
    }
}

spinner監察獲得所選項目的文字
spinner.onItemSelectedListener = object : AdapterView.OnItemSelectedListener{
    override fun onItemSelected(
        parent: AdapterView<*>?,
        view: View?,
        position: Int,
        id: Long
    ) {
        val textView:String = parent?.findViewById<TextView>(R.id.name)?.text.toString()
        Toast.makeText(this@MainActivity, textView, Toast.LENGTH_SHORT).show()
    }

    override fun onNothingSelected(parent: AdapterView<*>?) {
    }
}
