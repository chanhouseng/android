class MainActivity : AppCompatActivity() {
    val pickPhoto = registerForActivityResult(ActivityResultContracts.PickVisualMedia()){uri ->
        if(uri != null){
            imageView.setImageURI(uri)
        }
    }
    private lateinit var imageView: ImageView
    private lateinit var originalBitmap: Bitmap
    private lateinit var mutableBitmap: Bitmap
    private lateinit var frameLayout: FrameLayout
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContentView(R.layout.activity_main)
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
            val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
            insets
        }

        --- start in here ---
        frameLayout = findViewById(R.id.frameLayout)
        imageView = findViewById(R.id.imageView)
        findViewById<Button>(R.id.button).setOnClickListener {
            pickPhoto.launch(PickVisualMediaRequest(ActivityResultContracts.PickVisualMedia.ImageOnly))
        }
        findViewById<Button>(R.id.button2).setOnClickListener {
            imageView.rotation = imageView.rotation + 90f
        }
        findViewById<Button>(R.id.button3).setOnClickListener {
            saveLayoutToGallery(frameLayout)
        }
    }
    fun saveLayoutToGallery(view: View){
        try {
            val bitmap = Bitmap.createBitmap(
                view.width,
                view.height,
                Bitmap.Config.ARGB_8888
            )

            val canvas = Canvas(bitmap)
            view.draw(canvas)

            saveImageToGallery(bitmap)
        }catch (e: Exception){
            Toast.makeText(this, "保存失敗: ${e.message}", Toast.LENGTH_SHORT).show()
        }
    }

    fun saveImageToGallery(bitmap: Bitmap){
        try {
            val fileName = "drawing_${System.currentTimeMillis()}.png"
            val contentValues = ContentValues().apply {
                put(MediaStore.Images.Media.DISPLAY_NAME, fileName)
                put(MediaStore.Images.Media.MIME_TYPE, "image/png")
                put(MediaStore.Images.Media.RELATIVE_PATH, Environment.DIRECTORY_PICTURES)
            }

            val uri = contentResolver.insert(
                MediaStore.Images.Media.EXTERNAL_CONTENT_URI,
                contentValues
            )

            uri?.let {
                contentResolver.openOutputStream(it)?.use { stream ->
                    bitmap.compress(Bitmap.CompressFormat.PNG, 100, stream)
                }
                true
            }?:false
            Toast.makeText(this, "Save to gallery", Toast.LENGTH_SHORT).show()
        }catch (e: Exception){
            Toast.makeText(this, e.message.toString(), Toast.LENGTH_SHORT).show()
        }
    }
}
