val bottomNavigationView: BottomNavigationView = findViewById(R.id.bottomNavigation)
val viewPager2 : ViewPager2 = findViewById(R.id.viewPager2)
viewPager2.adapter = viewAdapter(this)
bottomNavigationView.setOnItemSelectedListener {
    when(it.itemId){
        R.id.one -> viewPager2.currentItem = 0
        R.id.two -> viewPager2.currentItem = 1
    }
    true
}
viewPager2.registerOnPageChangeCallback(object : ViewPager2.OnPageChangeCallback(){
    override fun onPageSelected(position: Int) {
        super.onPageSelected(position)
        when(position){
            0 -> bottomNavigationView.selectedItemId = R.id.one
            1 -> bottomNavigationView.selectedItemId = R.id.two
        }
    }
})


        
val prefs = activity.getSharedPreferences("data", MODE_PRIVATE)
Data.checkSet.addAll(prefs.getStringSet("dataSet", emptySet<String>())?:emptySet())
linearLayout = view.findViewById(R.id.linearLayout)
for(data in Data.dataList){
    val layout = activity.layoutInflater.inflate(R.layout.product, null)
    layout.findViewById<TextView>(R.id.name).text = data.name
    if(Data.checkSet.contains(data.id.toString())){
        layout.findViewById<ImageView>(R.id.imageView).setImageResource(R.drawable.like)
    }
    else{
        layout.findViewById<ImageView>(R.id.imageView).setImageResource(R.drawable.unlike)
    }
    layout.setOnClickListener {
        if(Data.checkSet.contains(data.id.toString())){
            Data.checkSet.remove(data.id.toString())
            layout.findViewById<ImageView>(R.id.imageView).setImageResource(R.drawable.unlike)
            save()
        }
        else{
            Data.checkSet.add(data.id.toString())
            layout.findViewById<ImageView>(R.id.imageView).setImageResource(R.drawable.like)
            save()
        }
    }
    linearLayout.addView(layout)
}

    fun save(){
        val editor = requireActivity()?.getSharedPreferences("data", MODE_PRIVATE)?.edit()
        editor?.putStringSet("dataSet", Data.checkSet)
        editor?.apply()
    }
        
override fun onResume() {
    super.onResume()
    val likeList = Data.dataList.filter { Data.checkSet.contains(it.id.toString()) }
    linearLayout.removeAllViews()
    for(like in likeList){
        val layout = activity?.layoutInflater?.inflate(R.layout.product, null)
        layout?.findViewById<TextView>(R.id.name)?.text = like.name
        layout?.findViewById<ImageView>(R.id.imageView)?.setImageResource(R.drawable.like)
        linearLayout.addView(layout)
    }
}
