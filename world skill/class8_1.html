class ProButton : LinearLayout {
    constructor(context: Context) : super(context)
    var valueProgress = 0
    private var button : Button
    private var progressBar: ProgressBar
    init {
        LayoutInflater.from(context).inflate(R.layout.pro_button, this, true)
        button = findViewById(R.id.button)
        progressBar = findViewById(R.id.progressBar)
        button.setOnClickListener {
            valueProgress ++
            addValue()
        }
    }

    fun addValue(){
        progressBar.progress = valueProgress
    }
}
        


class Table : LinearLayout{
    constructor(context: Context) : super(context)

    private var textView : TextView
    private var progress: ProgressBar

    init {
        LayoutInflater.from(context).inflate(R.layout.table, this, true)
        textView = findViewById(R.id.tableTextView)
        progress = findViewById(R.id.tableProgressBar)
    }

    fun setName(name:String){
        textView.text = name
    }

    fun setValue(value: Int){
        progress.progress = value
    }
}
