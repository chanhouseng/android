<com.google.android.material.progressindicator.CircularProgressIndicator
    android:layout_width="match_parent"
    android:layout_height="match_parent"
    app:indicatorSize="100dp"    //設置圓形進度條的直徑大小
    app:trackColor="#999"      //設置整條的顏色
    app:indicatorColor="#18abc8"    //設置progress的顏色
    android:max="100"
    android:progress="20"
    app:trackThickness="15dp"    //設置progress的粗度
    app:trackCornerRadius="0dp"    //自帶會有圖角，可設定圖角
    app:indicatorInset="0dp"    //設置progress與整個layout間的距離(類似padding) 但不影大小
    android:indeterminate="false"/>    //false會固定。true是無限旋轉(加載中)
        


<com.google.android.material.progressindicator.LinearProgressIndicator
    android:layout_width="wrap_content"
    android:layout_height="wrap_content"
    android:layout_margin="10dp"
    app:trackThickness="10dp"    //設置粗度
    app:trackColor="#999"      //設罝整條的顏色

    android:max="100"
    android:progress="20"
    app:trackCornerRadius="0dp"    //自帶會有圖角，可設定圖角
    app:indicatorDirectionLinear="rightToLeft"/>    //設置progress的方向
