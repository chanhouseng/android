val main : LinearLayout = findViewById(R.id.main)
val imageView : ImageView = findViewById(R.id.imageView)

var scaleFactor = 1f
val scaleDetector = ScaleGestureDetector(this, object : ScaleGestureDetector.SimpleOnScaleGestureListener() {    //建立一個「縮放偵測器」
    override fun onScale(detector: ScaleGestureDetector): Boolean {
        val scale = detector.scaleFactor    //取得這一瞬間兩指的縮放比例
        val nextScale = scale * scaleFactor
        if(nextScale in (1f..5f)){
            scaleFactor = nextScale
            imageView.scaleX = nextScale    //把圖片的寬度和高度都縮放到新的倍數
            imageView.scaleY = nextScale
            imageView.pivotX = detector.focusX    //把縮放的中心點設成兩根手指的中間位置。
            imageView.pivotY = detector.focusY
        }
        return true
    }
})
        
val gestureDetector = GestureDetector(this, object : GestureDetector.SimpleOnGestureListener(){    //建立一個「手勢偵測器」
    override fun onScroll(
        e1: MotionEvent?,
        e2: MotionEvent,
        distanceX: Float,
        distanceY: Float
    ): Boolean {
        imageView.translationX -= distanceX    //distanceX / distanceY 是這一瞬間手指移動的距離
        imageView.translationY -= distanceY
        return true
    }
})
        
main.setOnTouchListener { v, event ->
    scaleDetector.onTouchEvent(event)    //把所有的觸控事件同時交給兩個偵測器處理。
    gestureDetector.onTouchEvent(event)
    true
}
