val time = LocalDateTime.now()    //獲得完整日期時間 (年/月/日/時/分/秒/納秒)
findViewById(R.id.dateTime).text = time.toString()

val date = LocalTime.now()    //獲得時間 (時/分/秒/納秒)
findViewById(R.id.localDate).text = date.toString()

val timestamp = System.currentTimeMillis()    //取得毫秒時間戳 可以計返日期
findViewById(R.id.timestamp).text = timestamp.toString()

val time2 = time.plusMonths(3)     //加三個月的時間
val formatter3 = DateTimeFormatter.ofPattern("dd MMMM yyyy, EEEE HH:mm")   //自定日期的顯示格式
findViewById(R.id.textView3).text = time2.format(formatter3)     //輸出
    
val dateString = "2024-10-06"
val dateChange = LocalDate.parse(dateString)       //字符串轉成時間 
val formatter5 = DateTimeFormatter.ofPattern("dd/MM/yyyy")
findViewById(R.id.textView4).text = dateChange.format(formatter5)

//val date3 = "12.30.2025"
//val time1 = LocalDate.of(date3.substring(7, 10).toInt(), date3.substring(1, 2).toInt(), date3.substring(4, 5).toInt())

val time4 = LocalDate.of(2021, 3, 20)
val time5 = LocalDate.of(2021, 3, 22)     //輸入日期得到日期的LocalDate的形式再進行處理
val formatter6 = DateTimeFormatter.ofPattern("yyyy-MM-dd")
findViewById(R.id.textView5).text = time4.format(formatter6)
findViewById(R.id.textView6).text = ChronoUnit.DAYS.between(time4, time5).toString()    //計算相差天數（精確到天）

val dateTime2 = LocalDateTime.ofInstant(Instant.ofEpochMilli(timestamp), ZoneId.systemDefault())
//LocalDateTime.ofInstant( ... ) -- 把()中的時間點，轉成本地日期時間
//Instant.ofEpochMilli(timestamp) -- 把「毫秒時間戳」轉成 時間瞬間（Instant）
//ZoneId.systemDefault() -- 使用手機目前設定的時區
findViewById(R.id.textView7).text = dateTime2.toString()
