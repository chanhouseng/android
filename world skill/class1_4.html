spinner = findViewById(R.id.spinner)

findViewById<Button>(R.id.button).setOnClickListener {
    val spinnerText = spinner.selectedItemId.toString()
    Toast.makeText(this, spinnerText, Toast.LENGTH_SHORT).show()
}
val region = listOf("china", "macau", "HK")
val adapter = ArrayAdapter(this, android.R.layout.simple_spinner_dropdown_item, region)  //設置spinner選項
val spinner2 = findViewById<Spinner>(R.id.spinner2)
spinner2.adapter = adapter
spinner2.onItemSelectedListener = object : AdapterView.OnItemSelectedListener{
    override fun onItemSelected(
        parent: AdapterView<*>?,
        view: View?,
        position: Int,
        id: Long
    ) {
        Toast.makeText(this@MainActivity, region[position], Toast.LENGTH_SHORT).show()
    }

    override fun onNothingSelected(parent: AdapterView<*>?) {

    }

}
