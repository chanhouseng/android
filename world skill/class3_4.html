val foodList: List = listOf(
    foodClass(1, "One", R.drawable.croissant),
    foodClass(2, "Two", R.drawable.gingerbread),
    foodClass(3, "Three", R.drawable.cheese_cake),
    foodClass(4, "One", R.drawable.croissant),
    foodClass(5, "Two", R.drawable.gingerbread),
    foodClass(6, "Three", R.drawable.cheese_cake),
    foodClass(7, "One", R.drawable.croissant),
    foodClass(8, "Two", R.drawable.gingerbread),
    foodClass(9, "Three", R.drawable.cheese_cake)
)
var clickSet = HashSet()
val linearLayout: LinearLayout = findViewById(R.id.linearLayout)
val prefs = getSharedPreferences("data", MODE_PRIVATE)
val prefsSet = prefs.getStringSet("foodClick", emptySet<String>()) ?: emptySet()  //攞返本地存嘅clicked List(HashSet)

for (food in foodList) {
    val layout = layoutInflater.inflate(R.layout.product, null)
    layout.findViewById<TextView>(R.id.textView).text = food.name
    layout.findViewById<ImageView>(R.id.imageView).setImageResource(food.img)
    if (prefsSet.contains(food.id.toString())) {   //將所有clicked的放入clickSet放便之後存入唔會覆蓋數據
        clickSet.add(food.id.toString())
        food.isClick = true
    }

    if (food.isClick) {
        layout.findViewById<TextView>(R.id.isClick).text = "Clicked"
    } else {
        layout.findViewById<TextView>(R.id.isClick).text = "Unclicked"
    }
    layout.setOnClickListener {
        clickSet.add(food.id.toString())
        layout.findViewById<TextView>(R.id.isClick).text = "Clicked"
        val editor = getSharedPreferences("data", MODE_PRIVATE).edit()
        editor.putStringSet("foodClick", clickSet)
        editor.apply()
    }
    linearLayout.addView(layout)
}
