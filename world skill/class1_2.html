日期選擇器
dataPickerDialog = DatePickerDialog(this)
findViewById<EditText>(R.id.editTextDate).setOnClickListener {
    dataPickerDialog.show()
    dataPickerDialog.setOnDateSetListener { _, year, month, dayOfMonth ->
        val dataString = String.format("%d-%02d-%02d", year, month + 1, dayOfMonth)
        Toast.makeText(this, dataString, Toast.LENGTH_SHORT).show()
        findViewById<EditText>(R.id.editTextDate).setText(dataString)
    }
}

editTime.setOnFocusChangeListener {_, hasFocus ->
    if(hasFocus){
        timePickerDialog.show()
    }
