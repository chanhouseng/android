        val intArray = intArrayOf(3, 5, 1, 2, 10, 7, 8, 9, 9, 1)
        var intArray2 = intArray.filter { it >= 5 }.toIntArray()  //filter之後會變list

        val array = arrayOf("aaas", "bs", "dzz", "c")

        array.sortBy { it.length }             //按字的長度排

        val nameArray = arrayOf("Daniel", "Jason", "Andy", "Joe")
        val nameNew = nameArray.filter { it.length >= 5 }.map {    //用filter篩選名字length>=5的, 再將佢地整體加*
            it + "*"
        }
        val nameNew = nameArray.map {      //如果唔夠5個長度, 就喺後面加*直到5 (padStart前面)
            it.padEnd(5, '*')
        }
        Toast.makeText(this, nameNew.joinToString(), Toast.LENGTH_SHORT).show()

ValueAnimator

        val valueAnnotation = ValueAnimator.ofInt(0, 100)                //// 建立從 0 到 100 的整數動畫
        valueAnnotation.duration = 2000                                  // 動畫持續2秒
        valueAnnotation.interpolator = DecelerateInterpolator()          // 減速插值器（越到後面越慢）
        var value = 0
        valueAnnotation.addUpdateListener {
            value = it.animatedValue as Int                              animatedValue 的值會從 0 平滑地變化到 100, 就將其俾值俾value, 再show到textView
            findViewById<TextView>(R.id.textView2).text = value.toString()
        }
        findViewById<TextView>(R.id.textView).setOnClickListener {
            valueAnnotation.start()                                      開始動畫
        }

        val valueAnim = ValueAnimator.ofInt(0, 300)
        valueAnim.duration = 1500
        var value = 0
        valueAnim.addUpdateListener {
            value = it.animatedValue as Int
            findViewById<LinearLayout>(R.id.linearLayout).layoutParams.height = value    //把 LinearLayout 的高度動態改成 value 的數值
            findViewById<LinearLayout>(R.id.linearLayout).requestLayout()                //通知 Android View 系統：「我的尺寸變了，請重新計算佈局並重繪。」
        }
        findViewById<TextView>(R.id.textView).setOnClickListener {
            valueAnim.start()
        }

滑動方向偵測

        var startX = 0f
        var startY = 0f
        val touchText = findViewById<TextView>(R.id.textView4)
        val linear : LinearLayout = findViewById(R.id.linear2)
        linear.setOnTouchListener { _, event ->
            when(event.action){
                MotionEvent.ACTION_UP -> {
                    startX = event.x
                    startY = event.y
                }

                MotionEvent.ACTION_DOWN -> {
                    val moveX = event.x
                    val moveY = event.y

                    val deltaX = moveX - startX
                    val deltaY = moveY - startY
                     if(Math.abs(deltaX) > Math.abs(deltaY)){
                         if(deltaX > 0){
                             touchText.text = "左滑"
                         }
                         else{
                             touchText.text = "右滑"
                         }
                     }
                    else{
                        if(deltaY > 0){
                            touchText.text = "下滑"
                        }
                         else{
                             touchText.text = "上滑"
                        }
                     }
                }
            }
            true
        }
