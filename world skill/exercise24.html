val textView : TextView = findViewById(R.id.textView)
    val textView2 : TextView = findViewById(R.id.textView2)
    val textView3 : TextView = findViewById(R.id.textView3)
    val textList = listOf(textView, textView2, textView3)
    var n = 1
    val scrollView2: HorizontalScrollView = findViewById(R.id.scrollView2)

    scrollView2.setOnScrollChangeListener { v, scrollX, scrollY, oldScrollX, oldScrollY ->
        var maxScroll = 700
        var scrollXNew = 0
        when{
            scrollX < 700 -> {
                n = 1
                scrollXNew = scrollX
            }
            scrollX in 700..<1400 -> {
                n = 2
                scrollXNew = scrollX -700
            }
            else -> n = 3
        }

        val alpha = (scrollXNew.toFloat()/maxScroll.toFloat()).coerceIn(0f, 1f)
        val tranY2 = (scrollXNew.toFloat()/ maxScroll.toFloat()).coerceIn(0f, 1f)*30
        if(n != 3){
            textList[n-1].translationY = -30 + tranY2
            textList[n-1].alpha = 1f-alpha
            textList[n].translationY = -tranY2
            textList[n].alpha = alpha
        }
    }
