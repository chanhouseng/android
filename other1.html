        fun showLog(){
        val custom = layoutInflater.inflate(R.layout.alerclos, null)
        val closeButton = custom.findViewById<ImageView>(R.id.imageView8)
        val editText = custom.findViewById<EditText>(R.id.editTextText3)
        val submit = custom.findViewById<Button>(R.id.button3)
        val dialogs = AlertDialog.Builder(activity).setView(custom).show()
        dialogs.window?.setBackgroundDrawable(ColorDrawable(Color.TRANSPARENT))  //出除alert的背景，變為透明
        dialogs.setCanceledOnTouchOutside(false)                                 //暗空白處唔會退出alert
        closeButton.setOnClickListener {
            dialogs.dismiss()
        }
        submit.setOnClickListener {
            if(editText.text.toString().trim() == ""){
                Toast.makeText(activity, "input", Toast.LENGTH_SHORT).show()
            }
            else{
                val editor = activity?.getSharedPreferences("think", MODE_PRIVATE)?.edit()
                editor?.putString("think", editText.text.toString())
                editor?.apply()
                dialogs.dismiss()
            }
        }
    }
