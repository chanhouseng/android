viewPager2.adapter = viewPager2_adatper(this)
viewPager2.registerOnPageChangeCallback(object : ViewPager2.OnPageChangeCallback(){      //當 ViewPager2 的滾動狀態或選中頁面改變時觸發。
    override fun onPageSelected(position: Int) {
        val root = supportFragmentManager.findFragmentByTag("f$position")?.view ?: return     //依據系統為 ViewPager2 頁面產生的預設標籤（格式為 "f" + 索引）來查找目前位置所對應的 Fragment 實例。
        root.post {
            val widthSpec = View.MeasureSpec.makeMeasureSpec(root.width, View.MeasureSpec.EXACTLY)    
            val heightSpec = View.MeasureSpec.makeMeasureSpec(0, View.MeasureSpec.UNSPECIFIED)     
            //View.measure() 這個方法決定了一個 View 要佔據多大空間。makeMeasureSpec(size, mode) size: 剛好給你這麼大的空間, mode: 精確、最大、未指定模式
            //root 已經有實際寬度了。但高度没有，View 自己計算要有多高，愛多高多高，我完全不管！
            root.measure(widthSpec, heightSpec)      系統會依據傳入的規格，重新計算這個 View 及其子 View 的尺寸。
            val h = root.measuredHeight     
            if(h > 0){
                viewPager2.layoutParams.height = h    //直接修改 ViewPager2 的佈局參數（LayoutParams）中的高度屬性。
                viewPager2.requestLayout()
            }
        }
    }
})
